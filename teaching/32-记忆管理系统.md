# 记忆管理系统 (Memory)

```python
class MemoryCompactionHook:
    """Hook for automatic memory compaction when context is full."""

    async def __call__(
        self,
        agent: ReActAgent,
        kwargs: dict[str, Any],
    ) -> dict[str, Any] | None:
        # 1. 重入保护
        if getattr(agent, self._REENTRANCY_ATTR, False):
            return None

        # 2. 获取token计数
        token_counter = get_token_counter(agent_config)
        str_token_count = await token_counter.count(...)

        left_compact_threshold = (
            running_config.memory_compact_threshold - str_token_count
        )

        # 3. 压缩工具结果
        if trc.enabled:
            await self.memory_manager.compact_tool_result(...)

        # 4. 检查是否需要压缩
        (messages_to_compact, _, is_valid) = await self.memory_manager.check_context(...)

        # 5. 执行压缩
        if running_config.context_compact.context_compact_enabled:
            compact_content = await self.memory_manager.compact_memory(...)
```

### compact_memory 方法

```python
async def compact_memory(
    self,
    messages: list[Msg],
    previous_summary: str = "",
    extra_instruction: str = "",
    **_kwargs,
) -> str:
    """Compact messages into a condensed summary."""
    self._prepare_model_formatter()
    agent_config = load_agent_config(self.agent_id)
    cc = agent_config.running.context_compact

    result = await self._reme.compact_memory(
        messages=messages,
        as_llm=self.chat_model,
        as_llm_formatter=self.formatter,
        as_token_counter=get_token_counter(agent_config),
        language=agent_config.language,
        max_input_length=agent_config.running.max_input_length,
        compact_ratio=cc.memory_compact_ratio,  # 默认0.75
        previous_summary=previous_summary,
        return_dict=True,
        add_thinking_block=cc.compact_with_thinking_block,
        extra_instruction=extra_instruction,
    )
    return result.get("history_compact", "")
```

```python
async def dream_memory(self, **kwargs) -> None:
    """Run one dream-based memory optimization."""
    logger.info("running dream-based memory optimization")

    # 创建备份
    self.backup_path = Path(self.working_dir).absolute() / "backup"
    self.backup_path.mkdir(parents=True, exist_ok=True)

    memory_file = Path(self.working_dir) / "MEMORY.md"
    if memory_file.exists():
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_file = self.backup_path / f"memory_backup_{timestamp}.md"
        shutil.copyfile(memory_file, backup_file)

    # 创建ReActAgent执行梦境任务
    dream_agent = ReActAgent(...)
```

**Dream提示词原则**:
- 极简去冗: 只保留"核心业务决策"、"确认的用户偏好"、"高价值可复用经验"
- 状态覆写: 用新状态替换旧状态
- 归纳整合: 将零碎规则提炼为通用条目
- 废弃剔除: 删除已证伪的假设

| `memory_compact_ratio` | 0.75 | 上下文达到75%时触发压缩 |

```python
def create_memory_search_tool(memory_manager):
    """Create a memory_search tool function with bound memory_manager."""

    async def memory_search(
        query: str,
        max_results: int = 5,
        min_score: float = 0.1,
    ) -> ToolResponse:
        """Search MEMORY.md and memory/*.md files semantically."""
        return await memory_manager.memory_search(
            query=query,
            max_results=max_results,
            min_score=min_score,
        )

    return memory_search
```
