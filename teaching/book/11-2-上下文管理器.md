## 1. 什么是上下文管理器

### 1.1 资源管理问题

读写文件时常见的问题：

```python
# ❌ 危险写法：忘记关闭文件
file = open("data.txt", "w")
file.write("Hello")
# 如果这中间发生异常，file 就不会关闭

# ✅ 老式写法：手动关闭
file = open("data.txt", "w")
try:
    file.write("Hello")
finally:
    file.close()  # 总是执行
```

### 1.2 解决方案：with 语句

```python
# ✅ with 语句自动管理资源
with open("data.txt", "w") as file:
    file.write("Hello")
# 自动关闭，无需手动 try/finally
```

### 1.3 原理

```
┌────────────────────────────────────┐
│           with 语句执行流程         │
├────────────────────────────────────┤
│                                    │
│  1. 调用 __enter__()               │
│  2. 执行代码块                      │
│  3. 调用 __exit__()（无论是否异常） │
│                                    │
└────────────────────────────────────┘
```

## 2. 类实现上下文管理器

### 2.1 基本结构

```python
class MyContext:
    """自定义上下文管理器"""

    def __enter__(self):
        """进入上下文时调用"""
        print(">>> 进入")
        return self  # 返回值绑定到 as 后面的变量

    def __exit__(self, exc_type, exc_val, exc_tb):
        """退出上下文时调用"""
        print(">>> 退出")
        # exc_type, exc_val, exc_tb 是异常信息
        # 返回 True 表示压制异常
        return False

# 使用
with MyContext() as ctx:
    print("执行中...")
```

**输出**：
```
>>> 进入
执行中...
>>> 退出
```

### 2.2 处理异常

```python
class SafeContext:
    """安全上下文管理器"""

    def __enter__(self):
        print(">>> 进入")
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        print(f">>> 退出 - 异常类型: {exc_type}")
        if exc_type is ValueError:
            print("捕获到 ValueError，压异常制")
            return True  # 压制异常
        return False  # 不压制异常

# 测试
with SafeContext() as ctx:
    print("正常执行")

with SafeContext() as ctx:
    raise ValueError("测试异常")
```

## 3. 文件操作示例

### 3.1 简化版

```python
class FileManager:
    """文件管理器"""

    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None

    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file:
            self.file.close()
        return False

# 使用
with FileManager("test.txt", "w") as f:
    f.write("Hello, World!")
```

### 3.2 完整版（带异常处理）

```python
class FileManager:
    """文件管理器 - 完整版"""

    def __init__(self, filename, mode="r"):
        self.filename = filename
        self.mode = mode
        self.file = None
        self.closed = False

    def __enter__(self):
        try:
            self.file = open(self.filename, self.mode)
            return self
        except IOError as e:
            raise IOError(f"无法打开文件 {self.filename}: {e}")

    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file and not self.file.closed:
            self.file.close()
        self.closed = True

        # 记录异常但不重新抛出
        if exc_type:
            print(f"异常被记录: {exc_val}")

        return False  # 不压制异常

    def read_all(self):
        if self.closed:
            raise ValueError("文件已关闭")
        return self.file.read()

# 使用
with FileManager("data.txt", "w") as fm:
    fm.file.write("Hello")
    # 文件在这里自动关闭
```

## 4. contextlib 简化

### 4.1 @contextmanager 装饰器

用生成器简化上下文管理器：

```python
from contextlib import contextmanager

@contextmanager
def my_context():
    """简化版上下文管理器"""
    print(">>> 进入")
    try:
        yield "返回值"  # 相当于 __enter__ 返回值
    finally:
        print(">>> 退出")  # 相当于 __exit__

# 使用
with my_context() as value:
    print(f"获得值: {value}")
```

### 4.2 完整示例

```python
from contextlib import contextmanager

@contextmanager
def timer():
    """计时器上下文管理器"""
    import time
    start = time.time()
    try:
        yield
    finally:
        duration = time.time() - start
        print(f"耗时: {duration:.3f}s")

# 使用
with timer():
    sum(range(1000000))
```

### 4.3 @asynccontextmanager

异步版本：

```python
from contextlib import asynccontextmanager

@asynccontextmanager
async def fetch_data(url):
    """异步获取数据"""
    import httpx
    async with httpx.AsyncClient() as client:
        response = await client.get(url)
        yield response.json()
```

## 5. 嵌套上下文管理器

### 5.1 嵌套使用

```python
class OuterContext:
    def __enter__(self):
        print("外层进入")
        return self
    def __exit__(self, *args):
        print("外层退出")

class InnerContext:
    def __enter__(self):
        print("内层进入")
        return self
    def __exit__(self, *args):
        print("内层退出")

# 嵌套
with OuterContext() as outer:
    with InnerContext() as inner:
        print("执行操作")
```

**输出**：
```
外层进入
内层进入
执行操作
内层退出
外层退出
```

### 5.2 合并写法

```python
with OuterContext() as outer, InnerContext() as inner:
    print("执行操作")
```

## 6. 实战：数据库连接管理

### 6.1 需求

创建数据库连接的上下文管理器。

### 6.2 实现

```python
from contextlib import contextmanager

class DatabaseConnection:
    """数据库连接"""

    def __init__(self, host, port, database):
        self.host = host
        self.port = port
        self.database = database
        self.connection = None

    def connect(self):
        """模拟连接"""
        print(f"连接到 {self.host}:{self.port}/{self.database}")
        self.connection = f"conn_{self.database}"
        return self.connection

    def disconnect(self):
        """断开连接"""
        print(f"断开连接: {self.connection}")
        self.connection = None

    def query(self, sql):
        """执行查询"""
        if not self.connection:
            raise ValueError("未连接数据库")
        return f"结果: {sql}"

@contextmanager
def db_transaction(host, port, database):
    """数据库事务上下文管理器"""
    conn = DatabaseConnection(host, port, database)
    conn.connect()

    try:
        yield conn
        print("提交事务")
    except Exception as e:
        print(f"回滚事务: {e}")
        raise
    finally:
        conn.disconnect()

# 使用
with db_transaction("localhost", 3306, "mydb") as db:
    result = db.query("SELECT * FROM users")
    print(result)
```

## 7. 小结

### with 语句语法

```python
with 表达式 [as 变量]:
    代码块

# 或多个
with A() as a, B() as b:
    代码块
```

### 上下文管理器协议

```python
class MyContext:
    def __enter__(self):
        """进入时执行，返回值绑定到 as"""
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        """退出时执行，return True 压制异常"""
        return False
```

### contextlib 简化

```python
from contextlib import contextmanager

@contextmanager
def my_context():
    try:
        yield 返回值
    finally:
        清理代码
```
