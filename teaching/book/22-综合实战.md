### 3.2 新闻获取器

```python
# daily_news/fetcher.py

import httpx
from typing import Dict, List, Any
from datetime import datetime

class NewsFetcher:
    """新闻获取器"""

    BASE_URL = "https://newsapi.org/v2"

    def __init__(self, api_key: str):
        self.api_key = api_key

    async def fetch_headlines(
        self,
        category: str = "technology",
        country: str = "cn",
        limit: int = 10
    ) -> List[Dict[str, Any]]:
        """获取新闻头条"""
        url = f"{self.BASE_URL}/top-headlines"

        params = {
            "category": category,
            "country": country,
            "pageSize": limit,
            "apiKey": self.api_key
        }

        try:
            async with httpx.AsyncClient(timeout=30.0) as client:
                response = await client.get(url, params=params)
                response.raise_for_status()
                data = response.json()

                if data.get("status") != "ok":
                    raise ValueError(f"API Error: {data.get('message')}")

                articles = data.get("articles", [])
                return [self._parse_article(a) for a in articles]

        except httpx.HTTPError as e:
            raise ConnectionError(f"Failed to fetch news: {e}")

    def _parse_article(self, article: Dict) -> Dict:
        """解析文章"""
        return {
            "title": article.get("title", ""),
            "description": article.get("description", ""),
            "source": article.get("source", {}).get("name", ""),
            "url": article.get("url", ""),
            "published_at": article.get("publishedAt", ""),
        }
```

### 3.3 内容解析器

```python
# daily_news/parser.py

from typing import List, Set
from datetime import datetime
from .fetcher import NewsFetcher

class NewsParser:
    """新闻解析器"""

    def __init__(self):
        self.seen_titles: Set[str] = set()

    def parse(
        self,
        articles: List[Dict],
        max_items: int = 5
    ) -> List[Dict]:
        """解析并过滤新闻"""
        parsed = []
        seen = set()

        for article in articles:
            # 跳过无标题或已见过的
            title = article.get("title", "").strip()
            if not title or title in seen:
                continue

            seen.add(title)

            # 清理数据
            cleaned = self._clean_article(article)
            parsed.append(cleaned)

            if len(parsed) >= max_items:
                break

        return parsed

    def _clean_article(self, article: Dict) -> Dict:
        """清理文章数据"""
        return {
            "title": article.get("title", "").strip(),
            "description": self._truncate(
                article.get("description", ""),
                max_length=150
            ),
            "source": article.get("source", ""),
            "url": article.get("url", ""),
            "summary": self._generate_summary(article),
        }

    def _truncate(self, text: str, max_length: int) -> str:
        """截断文本"""
        if not text:
            return ""
        if len(text) <= max_length:
            return text
        return text[:max_length].rsplit(" ", 1)[0] + "..."

    def _generate_summary(self, article: Dict) -> str:
        """生成一句话总结"""
        title = article.get("title", "")
        description = article.get("description", "")

        if description:
            return description[:100]
        return title[:80]
```

### 3.4 格式化器

```python
# daily_news/formatter.py

from typing import List
from datetime import datetime

class NewsFormatter:
    """新闻格式化器"""

    def format_daily(
        self,
        articles: List[Dict],
        category: str,
        date: datetime
    ) -> str:
        """格式化每日资讯"""
        header = self._format_header(category, date)
        items = self._format_items(articles)
        footer = self._format_footer(len(articles))

        return f"{header}\n\n{items}\n\n{footer}"

    def _format_header(self, category: str, date: datetime) -> str:
        """格式化头部"""
        category_names = {
            "tech": "科技",
            "business": "商业",
            "entertainment": "娱乐",
        }
        category_text = category_names.get(category, category)
        date_text = date.strftime("%Y年%m月%d日")

        return f"📰 每日资讯 ({category_text}) - {date_text}"

    def _format_items(self, articles: List[Dict]) -> str:
        """格式化条目"""
        lines = []
        for i, article in enumerate(articles, 1):
            lines.append(f"{i}. 【{article['source']}】{article['title']}")
            lines.append(f"   {article['description']}")
            lines.append(f"   🔗 {article['url']}")
            lines.append("")

        return "\n".join(lines)

    def _format_footer(self, count: int) -> str:
        """格式化尾部"""
        return f"共 {count} 条资讯\n🌐 由 QwenPaw 整理"
```

### 3.5 技能主类

```python
# daily_news/skill.py

from typing import Dict, Any, List
from datetime import datetime
from pydantic import ValidationError

from .config import DailyNewsConfig
from .fetcher import NewsFetcher
from .parser import NewsParser
from .formatter import NewsFormatter

class DailyNewsSkill:
    """每日资讯技能"""

    name = "daily_news"
    description = "获取每日资讯摘要"

    def __init__(self, config: Dict[str, Any]):
        """初始化技能"""
        try:
            self.config = DailyNewsConfig(**config)
        except ValidationError as e:
            raise ValueError(f"Invalid config: {e}")

        self.fetcher = NewsFetcher(self.config.api_key or "")
        self.parser = NewsParser()
        self.formatter = NewsFormatter()

    async def execute(self, params: Dict[str, Any]) -> str:
        """执行技能"""
        # 获取参数
        category = params.get("category", self.config.category)
        max_items = params.get("max_items", self.config.max_items)

        try:
            # 1. 获取新闻
            articles = await self.fetcher.fetch_headlines(
                category=category,
                limit=max_items
            )

            # 2. 解析新闻
            parsed = self.parser.parse(articles, max_items=max_items)

            if not parsed:
                return "今日暂无资讯更新"

            # 3. 格式化输出
            result = self.formatter.format_daily(
                articles=parsed,
                category=category,
                date=datetime.now()
            )

            return result

        except ConnectionError as e:
            return f"获取资讯失败: {e}"
        except Exception as e:
            return f"处理资讯时出错: {e}"

    def get_schema(self) -> Dict[str, Any]:
        """返回技能 schema"""
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    "category": {
                        "type": "string",
                        "description": "新闻分类",
                        "enum": ["tech", "business", "entertainment"]
                    },
                    "max_items": {
                        "type": "integer",
                        "description": "获取数量",
                        "minimum": 1,
                        "maximum": 10
                    }
                }
            }
        }
```
