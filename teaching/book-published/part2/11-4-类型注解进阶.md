## 1. 泛型基础

### 1.1 什么是泛型

泛型允许你编写可复用的代码，同时保持类型安全：

```python
# ❌ 没有泛型：类型丢失
def first_element(lst):
    return lst[0]

# ✅ 有泛型：保留类型信息
def first_element(lst: list[int]) -> int:
    return lst[0]
```

### 1.2 TypeVar

使用 `TypeVar` 定义可变类型：

```python
from typing import TypeVar

T = TypeVar('T')

def identity(x: T) -> T:
    """返回相同的值"""
    return x

# 使用
result1 = identity(42)       # int
result2 = identity("hello")  # str
result3 = identity([1, 2])   # list[int]
```

### 1.3 泛型类

```python
from typing import TypeVar

T = TypeVar('T')

class Stack(Generic[T]):
    """泛型栈"""

    def __init__(self):
        self._items: list[T] = []

    def push(self, item: T) -> None:
        self._items.append(item)

    def pop(self) -> T:
        if not self._items:
            raise IndexError("栈为空")
        return self._items.pop()

    def peek(self) -> T:
        return self._items[-1]

# 使用
int_stack: Stack[int] = Stack()
int_stack.push(1)
int_stack.push(2)
print(int_stack.pop())  # 2

str_stack: Stack[str] = Stack()
str_stack.push("hello")
print(str_stack.pop())  # hello
```

## 2. Protocol

### 2.1 结构子类型

Protocol 定义**结构化类型**（类似 Go 的接口）：

```python
from typing import Protocol, runtime_checkable

@runtime_checkable
class Readable(Protocol):
    """可读协议"""
    def read(self) -> str:
        ...

@runtime_checkable
class Writeable(Protocol):
    """可写协议"""
    def write(self, data: str) -> None:
        ...

# 文件对象同时满足两个协议
f = open("test.txt", "w")
print(isinstance(f, Readable))  # True
print(isinstance(f, Writeable))  # True
f.close()
```

### 2.2 Protocol 定义

```python
from typing import Protocol, runtime_checkable

class Sized(Protocol):
    """有大小的对象"""
    def __len__(self) -> int:
        ...

@runtime_checkable
class Drawable(Protocol):
    """可绘制的对象"""
    def draw(self) -> None:
        ...

class Circle:
    def __init__(self, radius: float):
        self.radius = radius

    def __len__(self) -> int:
        return int(self.radius * 2 * 3.14)  # 周长

    def draw(self) -> None:
        print(f"绘制圆，半径 {self.radius}")

def render(obj: Sized & Drawable) -> None:
    """渲染同时满足 Sized 和 Drawable 的对象"""
    obj.draw()
    print(f"大小: {len(obj)}")

circle = Circle(5)
render(circle)
```

### 2.3 内置 Protocol

Python 3.8+ 提供许多内置 Protocol：

```python
from typing import Iterator, Iterable, Callable

# 检测迭代器
def process_items(items: Iterable[int]) -> None:
    for item in items:
        print(item)

# 检测可调用
def apply(func: Callable[[int], int], value: int) -> int:
    return func(value)

result = apply(lambda x: x * 2, 5)  # 10
```

## 3. Callable 和函数类型

### 3.1 Callable 类型

```python
from typing import Callable

# 简单函数类型
def apply_twice(func: Callable[[int], int], x: int) -> int:
    return func(func(x))

result = apply_twice(lambda x: x + 1, 0)  # 2

# 无返回值的函数
def for_each(func: Callable[[int], None], items: list[int]) -> None:
    for item in items:
        func(item)

for_each(print, [1, 2, 3])

# 多参数函数
def reduce_(func: Callable[[int, int], int], items: list[int]) -> int:
    result = items[0]
    for item in items[1:]:
        result = func(result, item)
    return result

result = reduce_(lambda a, b: a + b, [1, 2, 3, 4])  # 10
```

### 3.2 可选参数和默认值

```python
from typing import Callable

def create_multiplier(factor: int, logger: Callable[[str], None] = print) -> Callable[[int], int]:
    """创建乘数函数"""
    def multiplier(x: int) -> int:
        result = x * factor
        logger(f"{x} * {factor} = {result}")
        return result
    return multiplier

# 使用默认 logger
double = create_multiplier(2)
print(double(5))  # 10

# 使用自定义 logger
import logging
logging.basicConfig(level=logging.WARNING)
double_with_log = create_multiplier(2, logger=logging.warning)
print(double_with_log(5))  # 10
```

### 3.3 复杂的 Callable

```python
from typing import Callable

# 返回函数的高阶函数
def create_logger(level: str) -> Callable[[str], Callable[[str], None]]:
    """创建日志函数"""
    def logger(message: str) -> None:
        print(f"[{level.upper()}] {message}")
    return logger

error_logger = create_logger("error")
error_logger("Something went wrong")

# 异步函数类型
from typing import Awaitable

async def fetch_data(url: str) -> bytes:
    import httpx
    async with httpx.AsyncClient() as client:
        response = await client.get(url)
        return response.content

async def process(async_func: Callable[[str], Awaitable[bytes]], url: str) -> int:
    data = await async_func(url)
    return len(data)

import asyncio
result = asyncio.run(process(fetch_data, "https://example.com"))
print(f"获取到 {result} 字节")
```

## 4. Union 和可选类型

### 4.1 Union 类型

```python
from typing import Union

def process(value: Union[int, str, float]) -> str:
    if isinstance(value, int):
        return f"整数: {value}"
    elif isinstance(value, str):
        return f"字符串: {value.upper()}"
    else:
        return f"浮点数: {value:.2f}"

print(process(42))      # 整数: 42
print(process("hello")) # 字符串: HELLO
print(process(3.14))     # 浮点数: 3.14
```

### 4.2 可选类型

```python
from typing import Optional

def find_user(user_id: int) -> Optional[dict]:
    """查找用户，不存在返回 None"""
    users = {1: {"name": "小明", "age": 25}}
    return users.get(user_id)

# 使用
user = find_user(1)
if user is not None:
    print(user["name"])
else:
    print("用户不存在")

# 短路求值
name = find_user(999).get("name") if find_user(999) else "匿名"
print(name)  # 匿名
```

### 4.3 链式 Optional

```python
from typing import Optional

def get_city(country: Optional[dict]) -> Optional[str]:
    """从国家信息中获取城市"""
    if country is None:
        return None
    capital = country.get("capital")
    if capital is None:
        return None
    return capital.get("name")

# 使用
country = {
    "name": "中国",
    "capital": {"name": "北京", "population": 2000}
}

city = get_city(country)
if city:
    print(f"首都: {city}")

# 使用 match (Python 3.10+)
def describe_city(country: Optional[dict]) -> str:
    match country:
        case {"capital": {"name": str(name), "population": int(pop)}}:
            return f"{name}，人口 {pop}万"
        case {"name": str(name)}:
            return f"{name}，首都未知"
        case None:
            return "国家信息为空"
        case _:
            return "未知格式"
```

## 5. 类型别名和 NewType

### 5.1 类型别名

```python
from typing import Union, List, Dict

# 简单别名
UserId = int
ProductId = int
OrderId = int

# 复杂类型别名
Coordinates = tuple[float, float]
Matrix = List[List[float]]
JSON = Union[dict, list, str, int, float, bool, None]

# 使用
def distance(p1: Coordinates, p2: Coordinates) -> float:
    import math
    return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

p1: Coordinates = (1.0, 2.0)
p2: Coordinates = (4.0, 6.0)
print(distance(p1, p2))  # 5.0
```

### 5.2 NewType

```python
from typing import NewType

UserId = NewType('UserId', int)
ProductId = NewType('ProductId', int)

def get_user(user_id: UserId) -> str:
    return f"用户 {user_id}"

# 类型检查器会捕获错误
user_id = UserId(123)
product_id = ProductId(456)

print(get_user(user_id))    # OK
# print(get_user(product_id))  # 类型错误！类型不匹配
```

### 5.3 类型守卫

```python
from typing import Union

def is_string(value: Union[str, int]) -> bool:
    """类型守卫"""
    return isinstance(value, str)

def process(value: Union[str, int]) -> str:
    if is_string(value):
        # 类型检查器知道这里是 str
        return value.upper()
    else:
        # 类型检查器知道这里是 int
        return str(value * 2)

print(process("hello"))  # HELLO
print(process(5))        # 10
```

## 6. 实战：类型安全的配置类

### 6.1 需求

创建一个类型安全的配置类。

### 6.2 实现

```python
from typing import Generic, TypeVar, Optional, Literal
from pydantic import BaseModel, Field

T = TypeVar('T')

class Config(BaseModel, Generic[T]):
    """泛型配置类"""

    value: T
    description: str = ""
    required: bool = True

    def validate_value(self, input_value: T) -> bool:
        """验证值"""
        if not self.required and input_value is None:
            return True
        return isinstance(input_value, type(self.value))

class StringConfig(Config[str]):
    """字符串配置"""
    pass

class IntConfig(Config[int]):
    """整数配置"""
    pass

class ListConfig(Config[list]):
    """列表配置"""
    pass

# 使用
string_cfg = StringConfig(value="default", description="名称")
int_cfg = IntConfig(value=10, description="数量", required=False)
list_cfg = ListConfig(value=[1, 2, 3], description="ID 列表")

# 验证
print(string_cfg.validate_value("hello"))  # True
print(string_cfg.validate_value(123))       # False
print(int_cfg.validate_value(None))        # True (因为 required=False)
print(list_cfg.validate_value([1, 2]))      # True
```

### 6.3 运行时类型检查

```python
from typing import get_origin, get_args, Union, Optional
import inspect

def runtime_check(value: any, expected_type: type) -> bool:
    """运行时类型检查"""
    origin = get_origin(expected_type)
    args = get_args(expected_type)

    # 处理 Optional (Union with None)
    if origin is Union:
        if type(None) in args:
            non_none_args = [a for a in args if a is not type(None)]
            if value is None:
                return True
            return runtime_check(value, Union[tuple(non_none_args)])

    # 处理 List[int] 这种
    if origin is list:
        if not isinstance(value, list):
            return False
        item_type = args[0] if args else any
        return all(runtime_check(item, item_type) for item in value)

    # 处理 Dict[str, int] 这种
    if origin is dict:
        if not isinstance(value, dict):
            return False
        key_type, val_type = args if args else (any, any)
        return all(
            runtime_check(k, key_type) and runtime_check(v, val_type)
            for k, v in value.items()
        )

    # 基本类型
    return isinstance(value, expected_type)

# 测试
from typing import List, Dict, Optional

print(runtime_check("hello", str))                    # True
print(runtime_check([1, 2, 3], List[int]))             # True
print(runtime_check([1, "a"], List[int]))              # False
print(runtime_check({"a": 1, "b": 2}, Dict[str, int]))# True
print(runtime_check(None, Optional[str]))              # True
print(runtime_check("test", Optional[str]))            # True
```

## 7. 小结

### 泛型

```python
from typing import TypeVar, Generic

T = TypeVar('T')

class Container(Generic[T]):
    def __init__(self, value: T):
        self.value = value
```

### Protocol

```python
from typing import Protocol, runtime_checkable

@runtime_checkable
class Readable(Protocol):
    def read(self) -> str: ...

# 使用
def process(obj: Readable) -> None:
    print(obj.read())
```

### Callable

```python
from typing import Callable

def apply(func: Callable[[int], int], x: int) -> int:
    return func(x)
```

### Union 和 Optional

```python
from typing import Union, Optional

def process(value: Union[int, str]) -> str:
    ...

def optional() -> Optional[str]:
    return None
```

### 类型别名

```python
from typing import TypeAlias

JSON: TypeAlias = Union[dict, list, str, int, float, bool, None]
UserId: TypeAlias = int
```
