    """心跳配置的哈希，用于变更检测。"""

    """检查配置变更并在需要时重载"""
    try:
        mtime = self._config_path.stat().st_mtime  # 检查文件 mtime
    except FileNotFoundError:

    try:
        agent_config = load_agent_config(self._agent_id)  # 重载配置

        logger.exception(...)

## 3. 渠道热重载

### 3.1 _apply_channel_changes

```python
# src/qwenpaw/app/agent_config_watcher.py:191
async def _apply_channel_changes(self, agent_config: Any) -> None:
    """diff 渠道并重载变更的渠道"""
    if not agent_config.channels:
        return

    new_hash = self._channels_hash(agent_config.channels)
    if new_hash == self._last_channels_hash:  # 无变更
        return

    # 遍历可用渠道，检测变更
    for name in get_available_channels():
        new_ch = getattr(new_channels, name, None)
        old_ch = getattr(old_channels, name, None)
        new_dump = self._channel_dump(new_ch)
        old_dump = self._channel_dump(old_ch)
        if new_dump != old_dump:
            await self._reload_one_channel(name, new_ch, new_channels, old_ch)
```

### 3.2 单渠道重载

```python
# src/qwenpaw/app/agent_config_watcher.py:162
async def _reload_one_channel(
    self, name: str, new_ch: Any, new_channels: Any, old_ch: Any
) -> None:
    """重载单个渠道，失败时回退"""
    try:
        old_channel = await self._channel_manager.get_channel(name)
        new_channel = old_channel.clone(new_ch)  # 用新配置克隆
        await self._channel_manager.replace_channel(new_channel)
        logger.info(f"channel '{name}' reloaded")
    except Exception:
        logger.exception(f"failed to reload channel '{name}'")
        setattr(new_channels, name, old_ch)  # 失败时回退
```

### 3.3 ChannelManager.replace_channel

```python
# src/qwenpaw/app/channels/manager.py:704
async def replace_channel(self, new_channel: BaseChannel) -> None:
    """按名称替换单个渠道"""
    # 1) 在锁外启动新渠道（可能较慢）
    await new_channel.start()

    # 2) 在锁内交换 + 停止旧渠道
    async with self._lock:
        for i, ch in enumerate(self.channels):
            if ch.channel == new_channel.channel:
                self.channels[i] = new_channel  # 原子替换
                await old_channel.stop()  # 优雅停止
```

async def _apply_heartbeat_change(self, agent_config: Any) -> None:

async def reschedule_heartbeat(self) -> None:
    """重载心跳配置并更新或移除心跳任务"""
    async with self._lock:
        hb = get_heartbeat_config(self._agent_id)  # 重载配置

        # 移除现有心跳任务
        if self._scheduler.get_job(HEARTBEAT_JOB_ID):
            self._scheduler.remove_job(HEARTBEAT_JOB_ID)

        # 如果启用则添加心跳任务
        if getattr(hb, "enabled", False):
            trigger = self._build_heartbeat_trigger(hb.every)
            self._scheduler.add_job(
                self._heartbeat_callback,
                trigger=trigger,
                id=HEARTBEAT_JOB_ID,
                replace_existing=True,
            )

## 6. Workspace 重建与服务复用

除了通过 AgentConfigWatcher 轮询触发的增量热重载外，QwenPaw 还支持通过 API 调用触发的 **Workspace 级别完整重建**。这种模式下，整个 Workspace 被替换，但可复用服务（如记忆、缓存）被保留并迁移到新 Workspace。

### 6.1 API 触发式热重载

通过 REST API 更新配置时，系统使用异步后台任务触发 Agent 重载，不阻塞 API 响应。

源码路径：`src/qwenpaw/app/utils.py:17`

```python
# src/qwenpaw/app/utils.py:17
def schedule_agent_reload(request: "Request", agent_id: str) -> None:
    """在后台调度 Agent 重载（非阻塞）。

    从请求上下文提取 manager 和 agent_id，避免在后台任务中访问
    已结束生命周期的 request/workspace 对象。
    """
    manager = request.app.state.multi_agent_manager

    async def reload_in_background():
        try:
            await manager.reload_agent(agent_id)
        except Exception as e:
            logger.warning(f"Background reload failed: {e}")

    asyncio.create_task(reload_in_background())
```

以下 API 操作会触发热重载：

| 操作 | 方法 | 触发重载 |
|------|------|----------|
| 更新所有渠道 | `PUT /api/config/channels` | 是 |
| 更新单个渠道 | `PUT /api/config/channels/{channel_name}` | 是 |
| 更新 ACP 配置 | `PUT /api/config/acp` | 是 |
| 更新心跳配置 | `PUT /api/config/heartbeat` | 是 |
| 更新安全配置 | `PUT /api/config/security/tool-guard` | 是 |

### 6.2 MultiAgentManager.reload_agent 流程

源码路径：`src/qwenpaw/app/multi_agent_manager.py`

当 `schedule_agent_reload` 触发后，`reload_agent` 执行以下 8 步重建流程：

```python
async def reload_agent(self, agent_id: str) -> None:
    # 1. 获取当前 agent 的 workspace
    workspace = self._workspaces.get(agent_id)
    if not workspace:
        raise ValueError(f"Agent {agent_id} not found")

    # 2. 获取可复用服务（停止旧 workspace 前）
    old_reusable = workspace.service_manager.get_reusable_services()

    # 3. 停止旧 workspace（非 final 服务）
    # final=True 会完全清理，final=False 保留复用
    await workspace.service_manager.stop_all(final=False)

    # 4. 重新加载配置
    agent_config = load_agent_config(agent_id)

    # 5. 重建 workspace
    new_workspace = await self._create_workspace(agent_id, agent_config)

    # 6. 复用可复用服务
    for name, instance in old_reusable.items():
        await new_workspace.service_manager.set_reusable(name, instance)

    # 7. 启动新 workspace
    await new_workspace.service_manager.start_all()

    # 8. 更新 workspaces 字典
    self._workspaces[agent_id] = new_workspace
```

### 6.3 Workspace 重建状态机

Workspace 重建过程遵循严格的状态机模型，确保每个阶段完成后才进入下一阶段：

```
┌─────────────┐
│   IDLE      │ ← 初始状态
└──────┬──────┘
       │ reload_agent()
       ▼
┌─────────────┐
│  STOPPING   │ ← 停止旧服务（final=False 保留可复用服务）
└──────┬──────┘
       │ stop_all() 完成
       ▼
┌─────────────┐
│  CREATING   │ ← 创建新 Workspace
└──────┬──────┘
       │ _create_workspace() 完成
       ▼
┌─────────────┐
│  REUSING    │ ← 复用服务（记忆、缓存等）
└──────┬──────┘
       │ set_reusable() 完成
       ▼
┌─────────────┐
│  STARTING   │ ← 启动新服务
└──────┬──────┘
       │ start_all() 完成
       ▼
┌─────────────┐
│   IDLE      │ ← 重载完成
└─────────────┘
```

### 6.4 服务复用机制

源码路径：`src/qwenpaw/app/workspace/service_manager.py:94`

`set_reusable` 方法将旧 Workspace 中的可复用服务迁移到新 Workspace：

```python
# src/qwenpaw/app/workspace/service_manager.py:94
async def set_reusable(self, name: str, instance: Any) -> None:
    """将服务标记为从旧 workspace 复用。

    如果 descriptor 有 reload_func，会被调用。
    """
    if name not in self.descriptors:
        return

    descriptor = self.descriptors[name]
    if not descriptor.reusable:
        return  # 非可复用服务，不处理

    self.services[name] = instance
    self.reused_services.add(name)

    # 调用 reload_func（如果提供）
    if descriptor.reload_func is not None:
        result = descriptor.reload_func(self.workspace, instance)
        if asyncio.iscoroutine(result):
            await result
```

**服务类型对比：**

| 服务类型 | reusable=True | reusable=False |
|----------|---------------|-----------------|
| MemoryManager | 复用记忆数据 | 重建，丢失未持久化数据 |
| 数据库连接 | 复用连接 | 重建连接 |
| 缓存 | 复用缓存 | 清空缓存 |
| Channel (telegram) | 不适用 | 完全重建 |

**reload_func 示例：** 可复用服务可以通过 `reload_func` 在迁移时刷新配置，无需重建实例：

```python
ServiceDescriptor(
    name="memory_manager",
    factory=create_memory_manager,
    reusable=True,  # 可复用
    reload_func=lambda ws, instance: instance.reload_config(ws.config)
)
```

## 11. 应用场景

### 11.1 Telegram Bot Token 轮换

**场景**：Telegram Bot Token 过期，需要更新而不重启服务。

**流程**：
1. 修改 `agent.json` 中的 `channels.telegram.token`
2. mtime 变化触发 `_check()`
3. `_apply_channel_changes()` 检测到变更
4. `_reload_one_channel()` 克隆并替换渠道
5. 新 Token 立即生效，无需重启

### 11.2 心跳频率调整

**场景**：修改心跳间隔从 60 秒到 30 秒。

**流程**：
1. 修改 `agent.json` 中的 `heartbeat.every`
2. `_apply_heartbeat_change()` 检测到哈希变化
3. `CronManager.reschedule_heartbeat()` 重建调度任务
4. 新频率立即生效

### 11.3 禁用/启用渠道

**场景**：临时禁用 Discord 渠道。

**流程**：
1. 在 `agent.json` 中设置 `channels.discord.enabled: false`
2. 热重载检测变更
3. 渠道优雅停止，不再接收消息

## 13. 常见问题

### Q1: 配置变更未检测到？

**排查**：
1. 确认 `agent.json` 文件的 mtime 确实变化
2. 检查 `AgentConfigWatcher` 是否启动
3. 确认 `poll_interval` 不是过大值
4. 查看 watcher 日志

### Q2: 渠道重载失败？

**常见原因**：
1. 新配置格式错误
2. 新渠道凭证无效
3. 网络连接问题

**解决方案**：查看错误日志确认失败原因，配置回退到旧值。

### Q3: 热重载影响正在处理的消息？

**说明**：不会。热重载只影响后续消息，已在队列中的消息继续处理。

### Q4: 如何强制触发热重载？

**方式**：
1. 手动更新 `agent.json` 的 mtime：`touch agent.json`
2. 调用 `/daemon reload-config` 命令

### Q5: 部分配置更新未生效？

**排查**：
1. 确认变更的部分有对应的热重载处理（如渠道、心跳）
2. 某些配置可能需要完整重启才能生效
3. 检查 `_apply_*` 方法是否覆盖该配置项

## 附录：进阶热重载策略

### Watchdog vs Polling 对比

当前实现使用 **轮询（Polling）** 方式检测配置变更。这是简单可靠的方案，但在高频变更场景下有延迟。进阶方案对比：

| 方案 | 原理 | 延迟 | 资源消耗 | 实现复杂度 |
|------|------|------|----------|-----------|
| Polling（当前） | 定时检查 mtime | 轮询间隔 | 低 | 简单 |
| inotify（Linux） | 内核事件通知 | 毫秒级 | 极低 | 中等 |
| FSEvents（macOS） | 系统文件事件 | 毫秒级 | 极低 | 中等 |
| Watchdog 库 | 跨平台封装 | 毫秒级 | 低 | 中等 |

**Python watchdog 库使用示例：**

```python
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class ConfigFileHandler(FileSystemEventHandler):
    def __init__(self, callback):
        self.callback = callback

    def on_modified(self, event):
        if not event.is_directory and event.src_path.endswith("agent.json"):
            self.callback()

# 使用
observer = Observer()
handler = ConfigFileHandler(on_config_change)
observer.schedule(handler, str(workspace_dir), recursive=False)
observer.start()
```

### 配置变更的幂等性处理

生产环境中，配置变更可能重复触发（如编辑器保存、git checkout）。热重载应该是 **幂等** 的：

```python
class AgentConfigWatcher:
    async def _check(self):
        mtime = self._config_path.stat().st_mtime
        if mtime == self._last_mtime:
            return  # 无变更，快速退出

        self._last_mtime = mtime
        new_config = load_agent_config(self._agent_id)

        # 对比哈希，只有真正变更才应用
        if self._config_hash(new_config) == self._last_config_hash:
            return

        await self._apply_changes(new_config)
```

### 配置回滚机制

生产级系统应支持配置回滚：

```python
class ConfigVersion:
    """配置版本快照"""
    config: AgentProfileConfig
    timestamp: datetime
    checksum: str

class AgentConfigWatcher:
    def __init__(self):
        self._history: list[ConfigVersion] = []
        self._max_history = 10

    async def _apply_changes(self, new_config):
        # 保存当前配置快照
        current = self._load_current_config()
        self._history.append(ConfigVersion(
            config=current,
            timestamp=datetime.now(timezone.utc),
            checksum=self._compute_checksum(current)
        ))

        # 限制历史长度
        if len(self._history) > self._max_history:
            self._history.pop(0)

        # 应用新配置
        await self._do_apply(new_config)

    async def rollback(self, steps: int = 1):
        """回滚到之前的配置"""
        if len(self._history) < steps:
            raise ValueError("No enough history for rollback")

        target = self._history[-steps]
        await self._do_apply(target.config)
        self._history = self._history[:-steps]
```

### 12.4 如何避免破坏架构

**哈希比较规范**：
- 使用 `hashlib.md5` 或 `hashlib.sha256` 计算哈希
- 只比较变化的字段，避免全量比较
- 哈希命中时应立即返回，不执行后续逻辑

**锁策略规范**：
- 启动操作放锁外，减少锁持有时间
- 交换操作放锁内，保证原子性
- 停止操作放锁内，避免并发停止问题
