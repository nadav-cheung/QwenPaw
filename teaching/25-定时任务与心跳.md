## 概述

QwenPaw 的定时任务系统基于 **APScheduler** 构建，支持：
- 标准 cron 表达式调度
- 间隔触发（每 N 分钟/小时）
- 定时心跳保持智能体活跃
- 定时记忆整理（Dream）

本章通过源码逐行解析，揭示定时任务的设计与实现。

```python
class CronManager:
    """定时任务管理器 — 基于 APScheduler.AsyncIOScheduler"""

    def __init__(
        self,
        *,
        repo: BaseJobRepository,
        runner: Any,
        channel_manager: Any,
        timezone: str = "UTC",
        agent_id: Optional[str] = None,

        self._runner = runner                    # AgentRunner 实例
        self._channel_manager = channel_manager  # 渠道管理器

        self._executor = CronExecutor(
            runner=runner,
            channel_manager=channel_manager,

        self._lock = asyncio.Lock()            # 线程安全锁
        self._states: Dict[str, CronJobState] = {}  # 运行时状态
        self._rt: Dict[str, _Runtime] = {}     # 每任务信号量
```

```python
# src/qwenpaw/app/crons/manager.py:58
async def start(self) -> None:
    """启动调度器 — 从仓库加载所有任务并注册"""
    jobs = await self._repo.load()             # 从 JSON 文件加载任务
    for spec in jobs.jobs:
        if not spec.enabled:
            continue
        try:
            await self._register_or_update(spec)
        except Exception:
            # 无效任务自动禁用并保存
            spec.enabled = False
            await self._repo.save(jobs)

    # 注册心跳任务（如果启用）
    hb_cfg = get_heartbeat_config(self._agent_id)
    if hb_cfg and hb_cfg.enabled:
        await self._register_heartbeat(hb_cfg)

    # 注册 Dream 记忆整理任务
    dream_cron = get_dream_cron(self._agent_id)
    if dream_cron:
        await self._register_dream(dream_cron)

    self._scheduler.start()                     # 启动 APScheduler
    self._started = True
```

```python
# src/qwenpaw/app/crons/manager.py:108
async def _register_or_update(self, spec: CronJobSpec) -> None:
    """注册或更新一个定时任务到 APScheduler"""
    # 1. 验证 cron 表达式（必须是5字段）
    if len(spec.schedule.cron.split()) != 5:
        raise ValueError(f"Invalid cron: {spec.schedule.cron}")

    # 2. 创建并发控制信号量
    self._rt[spec.id] = _Runtime(
        sem=asyncio.Semaphore(spec.runtime.max_concurrency),
    )

    # 3. 注册到 APScheduler
    self._scheduler.add_job(
        func=self._execute_once,               # 执行函数
        trigger=CronTrigger.from_cron(spec.schedule.cron),  # cron 触发器
        args=(spec,),                           # 任务规格
        id=spec.id,                             # 任务 ID
        replace_existing=True,                  # 替换已存在任务
        misfire_grace_time=spec.runtime.misfire_grace_seconds,
        timezone=spec.schedule.timezone,
    )
```

```python
# src/qwenpaw/app/crons/manager.py:195
async def _execute_once(self, spec: CronJobSpec) -> None:
    """执行一个定时任务（由 APScheduler 调用）"""
    rt = self._rt.get(spec.id)
    if rt is None:
        return

    # 获取信号量（控制并发）
    if not rt.sem.locked():
        async with rt.sem:
            await self._do_execute(spec)
    else:
        # 任务已在运行，跳过本次触发
        rt.state.last_status = "skipped"
```

```python
async def create_or_replace_job(self, spec: CronJobSpec) -> None:
    async with self._lock:
        jobs = await self._repo.load()
        # 替换或添加
        jobs.jobs = [j for j in jobs.jobs if j.id != spec.id]
        jobs.jobs.append(spec)
        await self._repo.save(jobs)
        if self._started and spec.enabled:
            await self._register_or_update(spec)

# 删除任务
async def delete_job(self, job_id: str) -> bool:
    self._scheduler.remove_job(job_id)          # 从调度器移除
    async with self._lock:
        jobs = await self._repo.load()
        jobs.jobs = [j for j in jobs.jobs if j.id != job_id]
        await self._repo.save(jobs)

# 手动触发（fire-and-forget）
async def run_job(self, job_id: str) -> None:
    spec = await self._repo.get_job(job_id)
    asyncio.create_task(self._do_execute(spec))
```

### 3.2 agent 类型任务

源码路径：`src/qwenpaw/app/crons/executor.py:54`

```python
async def execute(self, job: CronJobSpec) -> None:
    """执行一个任务。

    - task_type text: 发送固定文本到渠道
    - task_type agent: 询问智能体并发送回复到渠道
      (stream_query + send_event)
    """
    target_user_id = job.dispatch.target.user_id
    target_session_id = job.dispatch.target.session_id
    dispatch_meta: Dict[str, Any] = dict(job.dispatch.meta or {})

    if job.task_type == "text" and job.text:
        await self._channel_manager.send_text(
            channel=job.dispatch.channel,
            user_id=target_user_id,
            session_id=target_session_id,
            text=job.text.strip(),
            meta=dispatch_meta,
        )
        return

    # agent: 作为调度目标用户运行请求以匹配上下文
    assert job.request is not None
    req: Dict[str, Any] = job.request.model_dump(mode="json")
    req["user_id"] = target_user_id or "cron"
    req["session_id"] = target_session_id or f"cron:{job.id}"

    async def _run() -> None:
        async for event in self._runner.stream_query(req):
            await self._channel_manager.send_event(
                channel=job.dispatch.channel,
                user_id=target_user_id,
                session_id=target_session_id,
                event=event,
                meta=dispatch_meta,
            )

    try:
        await asyncio.wait_for(
            _run(),
            timeout=job.runtime.timeout_seconds,
        )
    except asyncio.TimeoutError:
        logger.warning(
            "cron execute: job_id=%s timed out after %ss",
            job.id,
            job.runtime.timeout_seconds,
        )
        raise
    except asyncio.CancelledError:
        logger.info("cron execute: job_id=%s cancelled", job.id)
        raise
```

## 4. 心跳机制 — Heartbeat

心跳不是独立机制，而是作为特殊任务注册到 CronManager。`CronManager.start()` 启动时通过 `get_heartbeat_config()` 读取配置，调用 `_register_heartbeat()` 将心跳任务注册到 APScheduler。支持间隔触发（`"30m"`、`"1h30m"`）和 Cron 表达式两种格式，可配置活跃时段限制（`active_hours`）。

## 6. 数据模型

### 6.0 crontab 星期转换 — _crontab_dow_to_name

源码路径：`src/qwenpaw/app/crons/models.py:16-40`

APScheduler v3 使用 ISO 8601 星期编号（0=周一 ... 6=周日），而标准 crontab 使用（0=周日 ... 6=周六）。系统通过 `normalize_cron_5_fields` 在验证时自动将 crontab 数字转换为缩写名称：

```python
# src/qwenpaw/app/crons/models.py:16-40
_CRONTAB_NUM_TO_NAME: dict[str, str] = {
    "0": "sun",  # crontab: 周日
    "1": "mon",
    "2": "tue",
    "3": "wed",
    "4": "thu",
    "5": "fri",
    "6": "sat",  # crontab: 周六
    "7": "sun",  # 也可表示周日（兼容性）
}

def _crontab_dow_to_name(field: str) -> str:
    """将 day-of-week 字段从 crontab 数字转换为缩写名称。

    处理：*、单值、逗号分隔列表、范围。
    已经是名称的值（mon, tue, ...）直接透传。
    """
    if field == "*":
        return field

    def _convert_token(tok: str) -> str:
        if "/" in tok:  # 处理步长，如 "*/2"
            base, step = tok.rsplit("/", 1)
            return f"{_convert_token(base)}/{step}"
        if "-" in tok:  # 处理范围，如 "1-3"
            parts = tok.split("-", 1)
            return "-".join(_CRONTAB_NUM_TO_NAME.get(p, p) for p in parts)
        return _CRONTAB_NUM_TO_NAME.get(tok, tok)  # 数字→名称 或 透传

    return ",".join(_convert_token(t) for t in field.split(","))


# ScheduleSpec 验证器
@field_validator("cron")
@classmethod
def normalize_cron_5_fields(cls, v: str) -> str:
    parts = [p for p in v.split() if p]
    if len(parts) == 5:
        # 5字段：转换第5个字段（星期）
        parts[4] = _crontab_dow_to_name(parts[4])
        return " ".join(parts)

    if len(parts) == 4:
        # 4字段：hour dom month dow → 补秒 + 转换星期
        hour, dom, month, dow = parts
        return f"0 {hour} {dom} {month} {_crontab_dow_to_name(dow)}"

    if len(parts) == 3:
        # 3字段：dom month dow → 补时分 + 转换星期
        dom, month, dow = parts
        return f"0 0 {dom} {month} {_crontab_dow_to_name(dow)}"

    # 6字段（带秒）或更少：拒绝
    raise ConfigurationException(
        message="cron must have 5 fields (or 4/3 fields that can be normalized)",
    )
```

**转换示例**：
| crontab 表达式 | 转换前 | 转换后 | 说明 |
|---------------|--------|--------|------|
| `0 9 * * 0` | 第5字段=0 | `0 9 * * sun` | crontab 0=周日 → sun |
| `0 9 * * 1-3` | 第5字段=1-3 | `0 9 * * mon-wed` | 范围转换 |
| `0 9 * * */2` | 第5字段=*/2 | `0 9 * * */2` | 步长透传（已是名称）|
| `0 9 * * mon` | 第5字段=mon | `0 9 * * mon` | 已是名称，透传 |

```python
# src/qwenpaw/app/crons/repo/json_repo.py:20
class JsonJobRepository(BaseJobRepository):
    """JSON 文件持久化 — 支持原子写入"""

    async def save(self, jobs_file: JobsFile) -> None:
        """原子写入：先写 .tmp，再 move"""
        tmp_path = self._path.with_suffix(self._path.suffix + ".tmp")
        payload = jobs_file.model_dump(mode="json")
        tmp_path.write_text(
            json.dumps(payload, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )
        shutil.move(str(tmp_path), str(self._path))  # 原子替换
```

**原子写入的意义**：避免在写入过程中程序崩溃导致文件损坏。

## 9. 设计模式总结

### 9.1 并发控制 — Semaphore per Job

每个任务有自己的 `asyncio.Semaphore`，控制同一任务的最大并发数：

```python
self._rt[spec.id] = _Runtime(
    sem=asyncio.Semaphore(spec.runtime.max_concurrency),
)

async with rt.sem:          # 获取信号量
    await self._do_execute(spec)
```

**为什么不全局锁**：不同任务应互不影响，Job A 的慢不应阻塞 Job B。

### 9.2 状态机

```
[触发]
   │
   ▼
[获取信号量]
   │
   ├─── 已锁定 ──► skipped（跳过）
   │
   └─── 可用 ──► running
                    │
            ┌───────┴───────┐
            ▼               ▼
         success          error
            │               │
            ▼               ▼
       last_error=None  last_error=repr(e)
            │               │
            └───────┬───────┘
                    ▼
              last_run_at=now
```

### 9.3 Fire-and-Forget

手动触发使用 `asyncio.create_task()`，不等待完成：

```python
async def run_job(self, job_id: str) -> None:
    spec = await self._repo.get_job(job_id)
    asyncio.create_task(self._do_execute(spec))  # 不 await
```

**意义**：触发后立即返回，任务在后台异步运行。

### 9.4 Misfire Handling

```python
self._scheduler.add_job(
    ...,
    misfire_grace_time=spec.runtime.misfire_grace_seconds,  # 60秒
)
```

**意义**：如果系统繁忙导致触发延迟 ±60 秒内仍执行，避免累积延迟。

### 场景1：每日推送天气

```json
{
  "name": "Daily Weather",
  "schedule": {"cron": "0 8 * * *", "timezone": "Asia/Shanghai"},
  "task_type": "agent",
  "request": {"query": "查询北京今天天气，用简洁语言回复"},
  "dispatch": {
    "channel": "telegram",
    "target": {"user_id": "alice"}
```

## 11. 常见问题

### Q1: cron 表达式支持哪些格式？

**A**: 支持 3、4、5 字段格式，自动补全：

| 字段数 | 示例 | 补全后 |
|--------|------|--------|
| 3 | `0 * * *` | `0 0 * * *` (每天) |
| 3 | `0 0 * *` | `0 0 1 * *` (每月) |
| 4 | `30 9 * * *` | `0 30 9 * * *` |
| 5 | `0 30 9 * * *` | 不变 |

### Q2: misfire_grace_seconds 是什么？

**A**: 触发器错过的宽限期。例如设置每分钟执行一次，如果系统在第 59 秒才处理，应该在下一分钟补执行还是跳过？

- `misfire_grace_seconds=60`：在 60 秒内的错过会补执行
- `misfire_grace_seconds=0`：立即跳过

### Q3: 如何避免重复执行？

**A**: 每个任务有独立的 `asyncio.Semaphore`，控制并发数：

```python
# max_concurrency=1 时，同一时间只能有一个执行实例
# 如果任务执行中收到下一次触发，会被跳过（status=skipped）
```

### Q4: 心跳和 cron 任务有什么区别？

**A**: 主要区别：

| 特性 | 心跳 | Cron 任务 |
|------|------|-----------|
| 用途 | 保持 Agent 活跃 | 定时执行特定任务 |
| 配置 | HEARTBEAT.md | jobs.json |
| 触发条件 | 间隔/Cron + 活跃时段 | Cron 表达式 |
| 发送内容 | HEARTBEAT.md 内容 | 用户定义 |

### Q5: 如何调试定时任务？

**A**: 方法：
1. 查看任务状态：`GET /cron/jobs/{job_id}/state`
2. 手动触发：`POST /cron/jobs/{job_id}/run`
3. 检查 jobs.json 持久化状态
4. 查看日志中的 `CronManager` 相关输出

## 13. 如果你来自 Java...

对于有 Java 背景的开发者，理 QwenPaw 的定时任务系统时可以通过以下类比来理解：

| Python 概念 | Java 对应 | 说明 |
|-------------|----------|------|
| `CronManager` | `ScheduledExecutorService` | 定时任务管理器 |
| `AsyncIOScheduler` | `ScheduledThreadPoolExecutor` | 基于线程池的调度器 |
| `CronTrigger.from_cron()` | `@Scheduled(cron = "...")` | Cron 表达式触发器 |
| `IntervalTrigger` | `@Scheduled(fixedDelay = ...)` | 固定间隔触发 |
| `misfire_grace_seconds` | `misfireHandlingInstructions` | 错过触发时的处理 |
| `max_concurrency` | `RejectedExecutionHandler` | 并发控制 |
| `JsonJobRepository` | `JDBCJobStore` (Quartz) | 任务持久化 |
| `HeartbeatConfig` | `@Scheduled(fixedRate)` + 外部服务 | 保持活跃机制 |
| `CronJobSpec` | `JobDetail` (Quartz) | 任务规格定义 |

### 关键架构差异

**1. 调度器实现**

Java `ScheduledThreadPoolExecutor`：
```java
ScheduledExecutorService scheduler =
    Executors.newScheduledThreadPool(4);

// 固定延迟任务
scheduler.scheduleWithFixedDelay(
    () -> task(),
    initialDelay,
    period,
    TimeUnit.SECONDS
);

// Cron 任务（需要 Quartz）
@Scheduled(cron = "0 0 * * * ?")
public void hourlyTask() { ... }
```

QwenPaw 基于 APScheduler 的 `AsyncIOScheduler`：
```python
scheduler = AsyncIOScheduler(timezone="UTC")
scheduler.add_job(
    func=self._execute_once,
    trigger=CronTrigger.from_cron("0 * * * *"),
    id="hourly_job",
    replace_existing=True,
)
scheduler.start()
```

**2. 任务持久化**

Java Quartz 使用数据库：
```java
JobDetail job = JobBuilder.newJob(MyJob.class)
    .withIdentity("myJob", "group1")
    .build();

Trigger trigger = TriggerBuilder.newTrigger()
    .withIdentity("myTrigger", "group1")
    .withSchedule(CronScheduleBuilder.cronSchedule("0 0 * * * ?"))
    .build();

scheduler.scheduleJob(job, trigger);
```

QwenPaw 使用 JSON 文件：
```python
# jobs.json 结构
{
    "jobs": [
        {
            "id": "hourly_job",
            "schedule": {"cron": "0 * * * *", "timezone": "UTC"},
            "runtime": {"max_concurrency": 1},
            "enabled": true
        }
    ]
}
```

**3. 心跳机制**

Java 通常通过外部调度或消息队列实现：
```java
@Scheduled(fixedRate = 30000)  // 30秒
public void heartbeat() {
    agent.ping();  // 保持会话活跃
}
```

QwenPaw 的心跳是 Agent 内置机制：
```python
# config.json
{
    "heartbeat": {
        "enabled": true,
        "every": "30m",
        "active_hours": {"start": "09:00", "end": "22:00"}
    }
}
```

**4. 异步 vs 阻塞**

Java `ScheduledExecutorService` 可以是异步的，但任务执行在线程池中：
```java
scheduler.submit(() -> {
    // 任务在后台线程执行
    doSomething();
});
```

QwenPaw 使用 `AsyncIOScheduler`，任务在 asyncio 事件循环中执行：
```python
async def _execute_once(self, spec):
    async with rt.sem:
        await self._do_execute(spec)  # 异步执行
```

### 迁移建议

如果你从 Java Quartz/Spring Scheduler 迁移到 QwenPaw：
- 忘掉 **数据库 JobStore**，QwenPaw 使用 **JSON 文件**
- 忘掉 **`@Scheduled` 注解**，使用 **CronManager API** 注册任务
- 忘掉 **Java Thread**，使用 **Python asyncio**
- 心跳是 QwenPaw 内置的，不需要外部调度
- misfire 处理使用 `misfire_grace_seconds` 而不是复杂的 `MisfireInstruction`

## ★ Insight ─────────────────────────────────────
**cron 表达式自动补全**：QwenPaw 支持3/4/5字段的cron表达式，自动补全为标准5字段格式。`0 * * * *`（5字段）等价于 `* * * *`（1字段），系统会自动添加 `0` 前缀。这是为什么 `@hourly` 可以写作 `0 * * * *` 或直接写作 `* * * *`。

**心跳与 Dream 共享 APScheduler**：心跳和 Dream 不是独立的任务，而是共享同一个 `AsyncIOScheduler` 实例。这样做的好处是减少进程开销，但意味着如果调度器崩溃，所有定时任务都会受影响。

**发现的问题**：`agent_config_watcher.py:263` 调用 `self._cron_manager.reschedule_memory()` 但正确方法名是 `reschedule_dream()`。这意味着通过 Web UI 修改 Dream cron 表达式后，记忆优化任务可能不会自动重新调度。
