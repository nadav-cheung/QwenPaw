class MultiAgentManager:
    """管理多个智能体工作空间。

    特性：
    - 延迟加载：工作空间仅在首次请求时创建

    """

### 0.2 get_agent 延迟加载与去重

源码路径：`src/qwenpaw/app/multi_agent_manager.py:42`

```python
# src/qwenpaw/app/multi_agent_manager.py:42
async def get_agent(self, agent_id: str) -> Workspace:
    """通过 ID 获取智能体工作空间（带去重的延迟加载）。

    如果工作空间不在内存中，则创建并启动。
    相同 agent_id 的多个并发调用者会被协调：
    第一个调用者创建工作空间，其他等待。

    锁仅在字典检查/变更时短暂持有，不在工作空间
    启动期间持有，允许并行智能体初始化。
    """
    # 快速路径：已加载（无需锁）
    if agent_id in self.agents:
        logger.debug(f"Returning cached agent: {agent_id}")
        return self.agents[agent_id]

    should_start = False
    event = None
    agent_ref = None

    async with self._lock:
        # 在锁下重新检查
        if agent_id in self.agents:
            return self.agents[agent_id]

        if agent_id in self._pending_starts:
            # 另一个任务正在启动此智能体；等待它
            event = self._pending_starts[agent_id]
        else:
            # 我们是第一个调用者 — 验证配置并声明启动
            config = load_config()
            if agent_id not in config.agents.profiles:
                raise ConfigurationException(...)
            agent_ref = config.agents.profiles[agent_id]
            event = asyncio.Event()
            self._pending_starts[agent_id] = event
            should_start = True

    if not should_start:
        # 等待正在进行的启动完成
        await event.wait()
        if agent_id in self.agents:
            return self.agents[agent_id]
        raise ConfigurationException(...)

    # 我们是启动者 — 在锁外创建以实现并行
    instance = Workspace(agent_id=agent_id, workspace_dir=agent_ref.workspace_dir)
    await instance.start()
    instance.set_manager(self)

    async with self._lock:
        self.agents[agent_id] = instance

    return instance
```

### 0.3 零停机重新加载

源码路径：`src/qwenpaw/app/multi_agent_manager.py:138`

```python
# src/qwenpaw/app/multi_agent_manager.py:138
async def _graceful_stop_old_instance(
    self, old_instance: Workspace, agent_id: str
) -> None:
    """检查活跃任务后优雅停止旧实例。

    如果存在活跃任务，在后台安排延迟清理。
    否则立即停止。
    """
    has_active = await old_instance.task_tracker.has_active_tasks()

    if has_active:
        # 活跃任务 — 在后台安排延迟清理
        async def delayed_cleanup():
            try:
                completed = await old_instance.task_tracker.wait_all_done(timeout=60.0)
                if completed:
                    logger.info(f"All tasks completed for old instance {agent_id}.")
                await old_instance.stop(final=False)
            except Exception as e:
                logger.warning(f"Error during delayed cleanup for {agent_id}: {e}.")

        cleanup_task = asyncio.create_task(delayed_cleanup())
        self._cleanup_tasks.add(cleanup_task)
        cleanup_task.add_done_callback(lambda t: self._cleanup_tasks.discard(t))
    else:
        # 无活跃任务 — 立即停止
        await old_instance.stop(final=False)
```

### 0.4 并行启动所有已配置智能体

源码路径：`src/qwenpaw/app/multi_agent_manager.py:454`

```python
# src/qwenpaw/app/multi_agent_manager.py:454
async def start_all_configured_agents(self) -> dict[str, bool]:
    """并发启动配置中定义的所有已启用智能体。

    仅启动 enabled=True 的智能体。
    禁用的智能体被跳过以节省资源。

    智能体真正并行启动：get_agent() 仅在字典检查时
    短暂持有管理器锁，在慢速工作空间初始化期间释放。

    返回：
        dict[str, bool]: agent_id 到成功状态的映射
    """
    config = load_config()
    enabled_agents = {
        agent_id: ref
        for agent_id, ref in config.agents.profiles.items()
        if getattr(ref, "enabled", True)
    }
    agent_ids = list(enabled_agents.keys())

    async def start_single_agent(agent_id: str) -> tuple[str, bool]:
        try:
            await self.get_agent(agent_id)
            return (agent_id, True)
        except Exception as e:
            logger.error(f"Failed to start agent {agent_id}: {e}")
            return (agent_id, False)

    # 真正并行：get_agent 在工作空间启动期间释放锁
    results = await asyncio.gather(
        *[start_single_agent(agent_id) for agent_id in agent_ids],
        return_exceptions=False,
    )
    return dict(results)
```

### 0.5 关键设计模式

| 模式 | 说明 |
|------|------|
| 双重检查锁定 | 先无锁检查，再在锁内重新检查，确保线程安全且高性能 |
| 延迟初始化 | 工作空间仅在首次请求时创建，避免预加载开销 |
| 细粒度锁 | 锁仅在字典操作时持有，不在工作空间创建期间持有 |
| 零停机重载 | 新实例立即服务请求，旧实例在后台等待任务完成后清理 |

```

└─────────────────┘
```

### 3.1 ACPService 服务单例

**源码路径**: `src/qwenpaw/agents/acp/service.py`

```python
class ACPService:
    """ACP 服务单例，管理所有 ACP 会话"""

    def __init__(self, *, config: ACPConfig):
        self.config = config
        self._lock = asyncio.Lock()
        self._sessions: Dict[tuple[str, str], _Conversation] = {}

    async def run_turn(
        self,
        *,
        chat_id: str,
        agent: str,
        prompt_blocks: list[dict[str, Any]],
        cwd: str,
        on_message: MessageHandler,
        restart: bool = False,
        require_existing: bool = False,
    ) -> dict[str, Any]:
        """Execute one conversation turn.

        Args:
            chat_id: 聊天 ID（用于标识对话）
            agent: 智能体标识符
            prompt_blocks: 提示块列表
            cwd: 工作目录
            on_message: 消息处理器
            restart: 是否重启会话
            require_existing: 是否要求会话已存在

        Returns:
            包含结果或错误的字典
        """
        conversation = await self._get_or_create_session(
            chat_id=chat_id,
            agent=agent,
            cwd=cwd,
            restart=restart,
            require_existing=require_existing,
        )

        async with conversation.turn_lock:
            # 检查是否有待处理的权限请求
            if conversation.client.pending_permission is not None:
                raise ACPSessionError("Session is waiting for permission")

            # 启动提示
            conversation.client.start_prompt(on_message)
            conversation.prompt_task = asyncio.create_task(
                conversation.conn.prompt(
                    session_id=conversation.acp_session_id,
                    prompt=self._prompt_blocks_to_models(prompt_blocks),
                ),
            )
            return await self._wait_for_prompt_outcome(conversation)
```

### 3.2 权限恢复

**`resume_permission` 方法** (`service.py` 第96-122行):

```python
async def resume_permission(
    self,
    *,
    acp_session_id: str,
    option_id: str,
    on_message: MessageHandler,
) -> dict:
    """恢复暂停的权限请求"""

    conversation = await self._find_session_by_acp_id(acp_session_id)

    # 恢复提示
    conversation.client.resume_prompt(on_message)
    conversation.client.resolve_permission(option_id)
    await conversation.client.emit_permission_resolved()

    return await self._wait_for_prompt_outcome(conversation)
```

**`resume_permission` 流程**:
```

         ├── conversation.client.resolve_permission(option_id)  解决权限
         └── conversation.client.emit_permission_resolved()  发送解决事件
         │

async def _open_conversation(
    self,
    *,
    chat_id: str,
    agent: str,
    cwd: str,
    agent_config: ACPAgentConfig,
) -> _Conversation:
    """打开新会话"""
    client = ACPHostedClient(
        agent_name=agent,
        agent_config=agent_config,
        cwd=cwd,
    )
    exit_stack = AsyncExitStack()

    # 1. 启动子进程
    conn, process = await exit_stack.enter_async_context(
        spawn_agent_process(
            client,
            agent_config.command,
            *agent_config.args,
            cwd=cwd,
            env={**os.environ, **agent_config.env},
            transport_kwargs={
                "limit": agent_config.stdio_buffer_limit_bytes,
            },
        ),
    )

    # 2. 初始化协议
    initialized = await conn.initialize(
        protocol_version=PROTOCOL_VERSION,
        capabilities=ClientCapabilities(),
        client_info=Implementation(
            name="qwenpaw-acp-service",
            version="0.1.0",
        ),
    )

    # 3. 创建新会话
    new_session = await conn.new_session(cwd=cwd)
    return _Conversation(
        chat_id=chat_id,
        agent=agent,
        acp_session_id=new_session.session_id,
        cwd=cwd,
        conn=conn,
        process=process,
        client=client,
        exit_stack=exit_stack,
        turn_lock=asyncio.Lock(),
    )
```

### 3.5 cancel_turn 取消机制

**`cancel_turn` 方法** (`service.py` 第156-189行):

```python
async def cancel_turn(self, *, chat_id: str, agent: str) -> bool:
    """取消正在运行的 turn"""
    conversation = await self.get_session(chat_id, agent)
    if conversation is None:
        return False

    prompt_task = conversation.prompt_task
    if prompt_task is None or prompt_task.done():
        return False

    # 尝试发送取消请求
    for _ in range(3):
        try:
            await conversation.conn.cancel(
                session_id=conversation.acp_session_id,
            )
        except Exception:
            return False

        try:
            await asyncio.wait_for(
                asyncio.shield(prompt_task),
                timeout=0.5,
            )
        except asyncio.TimeoutError:
            if prompt_task.done():
                break
            continue
        except Exception:
            break
        else:
            break

    return prompt_task.done()
```

### 6.1 主从模式

```
┌──────────────┐         委托          ┌──────────────┐
│   Master     │ ─────────────────────►│   Worker     │
│   Agent      │ ◄─────────────────────│   Agent      │
│  (任务分解)   │         结果          │  (执行任务)   │
└──────────────┘                       └──────────────┘
```

```python
# 主从协作示例
async def master_worker_collaboration(task: str):
    # 1. 主智能体分解任务
    subtasks = await master.decompose_task(task)
    
    # 2. 并行委托给工作智能体
    results = await asyncio.gather(*[
        delegate_external_agent("worker_1", subtask)
        for subtask in subtasks
    ])
    
    # 3. 主智能体汇总结果
    final_result = await master.aggregate_results(results)
    
    return final_result
```

## 18. 知识检查

### 题目一

MultiAgentManager 的 `get_agent` 方法如何处理对同一 agent_id 的多个并发请求？这种设计模式叫什么？

<details>
<summary>参考答案</summary>

采用双重检查锁定（Double-Checked Locking）模式配合 `_pending_starts` 字典实现去重。第一次检查在锁外进行快速路径判断；如果未命中，在锁内重新检查。对于同一 agent_id 的并发请求，第一个调用者会创建 `asyncio.Event` 并记录到 `_pending_starts`，后续调用者发现已有 pending 状态后会等待该 Event。这样既保证了线程安全，又避免了重复创建工作空间，同时锁只在字典检查/变更时短暂持有，不会阻塞在工作空间启动期间。
</details>

### 题目二

ACP 协议中的 SuspendedPermission 机制解决了什么问题？如果去掉这个机制，智能体的行为会有什么变化？

<details>
<summary>参考答案</summary>

SuspendedPermission 解决的是外部智能体执行敏感操作时的用户确认问题。当外部 ACP 智能体（如 Claude Code）需要执行文件写入、命令执行等敏感操作时，会发出权限请求。ACPHostedClient 将其封装为 SuspendedPermission 并挂起当前 turn，等待用户通过 /approve 或 /deny 响应。

如果去掉这个机制，外部智能体的所有操作将自动执行，没有任何人工审批环节。这会导致安全风险：恶意或错误的工具调用可能直接修改文件系统、执行危险命令等，而用户毫不知情。
</details>

### 题目三

`_graceful_stop_old_instance` 方法中的零停机重载是如何实现的？为什么旧实例的清理不在锁内进行？

<details>
<summary>参考答案</summary>

零停机重载的实现方式是：先创建并启动新实例，然后立即将其放入 `self.agents` 字典，使新请求路由到新实例。旧实例的停止在锁外异步执行：

- 如果旧实例没有活跃任务，立即调用 `stop()`。
- 如果有活跃任务，启动一个后台 `delayed_cleanup` 协程，等待任务完成（最多 60 秒）后再停止旧实例。

旧实例的清理不在锁内进行，因为 `stop()` 和等待任务完成都可能耗时较长。如果在锁内等待，所有其他智能体的启动和访问请求都会被阻塞，造成系统级延迟。将清理移到锁外，确保管理器锁只用于快速的字典操作。
</details>
