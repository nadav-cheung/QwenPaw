```python
# Characters forbidden in Windows filenames
_UNSAFE_FILENAME_RE = re.compile(r'[\\/:*?"<>|]')

def sanitize_filename(name: str) -> str:
    """Replace characters that are illegal in Windows filenames with ``--``.

    >>> sanitize_filename('discord:dm:12345')
    'discord--dm--12345'
    >>> sanitize_filename('normal-name')
    'normal-name'
    """
    return _UNSAFE_FILENAME_RE.sub("--", name)
```

### 1.4 文件路径生成

**源码路径**: `src/qwenpaw/app/runner/session.py:97`

```python
def _get_save_path(self, session_id: str, user_id: str) -> str:
    """Return a filesystem-safe save path."""
    os.makedirs(self.save_dir, exist_ok=True)
    safe_sid = sanitize_filename(session_id)
    safe_uid = sanitize_filename(user_id) if user_id else ""
    if safe_uid:
        file_path = f"{safe_uid}_{safe_sid}.json"
    else:
        file_path = f"{safe_sid}.json"
    return os.path.join(self.save_dir, file_path)
```

| user_id | session_id | 文件名 |
|---------|------------|--------|
| `alice` | `telegram:chat_123` | `alice_telegram--chat_123.json` |
| `""` | `console:user1` | `console--user1.json` |

```python
async def save_session_state(
    self,
    session_id: str,
    user_id: str = "",
    **state_modules_mapping,
) -> None:
    """Save state modules to a JSON file using async I/O."""
    state_dicts = {
        name: state_module.state_dict()
        for name, state_module in state_modules_mapping.items()
    }
    session_save_path = self._get_save_path(session_id, user_id=user_id)
    with open(session_save_path, "w", encoding="utf-8") as f:
        f.write(json.dumps(state_dicts, ensure_ascii=False))
```

### 2.2 load_session_state

**源码路径**: `src/qwenpaw/app/runner/session.py:136`

```python
async def load_session_state(
    self,
    session_id: str,
    user_id: str = "",
    allow_not_exist: bool = True,
    **state_modules_mapping,
) -> None:
    """Load state modules from a JSON file using async I/O."""
    session_save_path = self._get_save_path(session_id, user_id=user_id)
    if os.path.exists(session_save_path):
        async with aiofiles.open(session_save_path, "r", encoding="utf-8") as f:
            content = await f.read()
            states = _safe_json_loads(content, session_save_path)

        for name, state_module in state_modules_mapping.items():
            if name in states:
                state_module.load_state_dict(states[name])
```

## 3. JSON 损坏恢复

**源码路径**: `src/qwenpaw/app/runner/session.py:24`

```python
def _safe_json_loads(content: str, filepath: str = "") -> dict:
    """Parse JSON with corruption recovery.

    Attempts standard ``json.loads`` first.  If that fails due to
    trailing garbage (a common symptom of concurrent-write race
    conditions), falls back to ``raw_decode`` to extract the first
    valid JSON object.
    """
    try:
        return json.loads(content)
    except json.JSONDecodeError:
        pass

    # Try to extract the first valid JSON object.
    try:
        result, _ = json.JSONDecoder().raw_decode(content)
        logger.warning(
            "Session file %s had corrupted JSON. "
            "Recovered first valid object via raw_decode.",
            filepath,
        )
        return result
    except json.JSONDecodeError:
        logger.warning(
            "Session file %s is completely corrupted and could not "
            "be recovered. Returning empty dict.",
            filepath,
        )
        return {}
```

**恢复机制**：

| 层级 | 策略 | 说明 |
|------|------|------|
| 1 | 标准解析 | `json.loads()` 首次尝试 |
| 2 | 截断恢复 | `raw_decode()` 提取第一个有效对象 |
| 3 | 返回空字典 | 完全损坏时返回 `{}` 并记录警告 |

**常见损坏原因**：

```python
# 1. 并发写入（Race Condition）
async def concurrent_write():
    # 两个协程同时写入同一文件
    await session.save_session_state(...)  # 协程 A
    await session.save_session_state(...)  # 协程 B
    # 可能产生: {"a": 1}{"b": 2}

# 2. 进程异常终止
# 写入一半时被 kill
# 可能产生: {"messages": [{"content": "hello

# 3. 网络文件系统延迟
# NFS/CIFS 写入延迟导致读到的不是最新数据
```

**源码路径**: `src/qwenpaw/app/workspace/service_manager.py:180`

```python
async def start_all(self) -> None:
    """Start all services by priority order.

    Same-priority services start concurrently.
    Different priorities start sequentially.
    """
    groups = defaultdict(list)
    for name, desc in self.descriptors.items():
        groups[desc.priority].append(desc)

    for priority in sorted(groups.keys()):
        descriptors = groups[priority]
        if all(d.concurrent_init for d in descriptors):
            # 并发启动同优先级的服务
            await asyncio.gather(*[
                self._start_service(desc) for desc in descriptors
            ])
        else:
            # 串行启动
            for desc in descriptors:
                await self._start_service(desc)
```

**并发启动示例**：

```
Priority 20: [memory_manager, mcp_manager, chat_manager]
            ↓
        asyncio.gather() 并发启动三者

Priority 25: [runner_start]
            ↓
        串行等待 runner_start 完成
```

## 7. 可复用组件模式

**源码路径**: `src/qwenpaw/app/workspace/workspace.py:280`

```python
async def set_reusable_components(self, components: dict) -> None:
    """设置可复用组件，支持热重载。

    必须在 start() 之前调用。
    """
    for name, component in components.items():
        await self._service_manager.set_reusable(name, component)
```

**热重载流程**:

```
旧 Workspace 停止
    │
    ├── 标记 memory_manager, chat_manager 为可复用
    │       ServiceManager.set_reusable("memory_manager", old_instance)
    │
    ▼
新 Workspace 创建
    │
    ├── set_reusable_components() 注入旧组件
    │
    ▼
新 Workspace 启动
    │
    └── ServiceManager 检测到是可复用组件
            │
            └── 跳过初始化，直接使用已有实例
```

**可复用组件的优势**：
1. **热重载速度提升**：不需要重新初始化 memory_manager
2. **内存节省**：多个 Workspace 共享同一个 memory_manager 实例
3. **状态保持**：重载后记忆不丢失

## 8. Workspace 生命周期

**源码路径**: `src/qwenpaw/app/workspace/workspace.py:325`

```python
async def start(self):
    """Start workspace and initialize all components."""
    if self._started:
        return

    # 1. 加载 Agent 配置
    self._config = load_agent_config(self.agent_id)

    # 2. 启动所有服务
    await self._service_manager.start_all()

    self._started = True

async def stop(self, final: bool = True):
    """Stop workspace and clean up all resources.

    Args:
        final: If True (default), stop ALL services.
               If False, skip reusable services (for reload).
    """
    if not self._started:
        return

    await self._service_manager.stop_all(final=final)
    self._started = False
```

**final 参数说明**：

| final 值 | 行为 |
|----------|------|
| `True` | 停止所有服务，包括可复用组件 |
| `False` | 保留可复用组件（用于热重载） |

## 10. 最佳实践

### 10.1 定期保存

```python
# 避免频繁写盘，设置合理的保存间隔
class SessionManager:
    def __init__(self, session, flush_interval=5.0):
        self.session = session
        self._buffer = {}
        self._last_save = time.time()
        self._flush_interval = flush_interval

    async def update(self, key, value):
        self._buffer[key] = value
        if time.time() - self._last_save > self._flush_interval:
            await self._flush()

    async def _flush(self):
        for key, value in self._buffer.items():
            await self.session.update_session_state(
                session_id=self.session_id,
                key=key,
                value=value
            )
        self._last_save = time.time()
```

## 11. 常见问题

### Q1: 会话文件损坏

**原因**：并发写入或进程异常终止。

**解决**：
```bash
# SafeJSONSession 自动尝试恢复
# 如果无法恢复，手动清理
rm sessions/corrupted_session.json

# 下次请求时会创建新的空会话
```

### Q2: 热重载后记忆丢失

**原因**：`stop(final=True)` 停止了可复用组件。

**解决**：
```python
# 热重载时使用 final=False
await workspace.stop(final=False)  # 保留 memory_manager
```

### Q3: 多 Workspace 共享状态

**原因**：错误地将同一组件实例用于多个 Workspace。

**解决**：
```python
# 每个 Workspace 应有独立的 memory_manager
# 除非明确设置为可复用组件
await workspace.set_reusable_components({
    "memory_manager": shared_memory_manager
})
```

### Q4: Windows 文件名不兼容

**原因**：session_id 包含 Windows 非法字符。

**解决**：`SafeJSONSession._get_save_path` 自动调用 `sanitize_filename` 处理。

## 来自 Java 的你

### 核心概念对照

| Java | Python / QwenPaw | 说明 |
|------|-------------------|------|
| `HttpSession` | `SafeJSONSession` | Java Servlet 的 HttpSession 是内存中的会话对象，SafeJSONSession 是基于 JSON 文件的持久化会话 |
| `ServletContext` | `AgentContext` / `Workspace` | ServletContext 是全局应用上下文，Workspace 是每个 Agent 的独立运行上下文 |
| `@SessionScoped` | 会话持久化 (`save_session_state`) | Java CDI 用 `@SessionScoped` 注解声明会话作用域 Bean，QwenPaw 通过显式 `save_session_state()` 调用持久化 |
| `HttpSessionListener` | Session hooks / `ServiceManager.start_all()` | Java 用 Listener 监听会话创建/销毁事件，QwenPaw 通过 ServiceManager 优先级管理服务生命周期 |
| Java 序列化 (`Serializable`) | JSON `json.dumps` / `json.loads` | Java 用内置序列化机制，QwenPaw 用 JSON 文本格式，并内置 `raw_decode` 损坏恢复 |

### 关键差异

Java HttpSession 存储在 Servlet 容器内存中（可配置为 Redis 等外部存储），由容器自动管理生命周期；SafeJSONSession 将状态持久化为磁盘上的 JSON 文件，需要手动调用 `save_session_state()` 和 `load_session_state()`。Java 开发者需要注意：这里没有容器的会话超时机制，JSON 文件损坏时通过 `raw_decode` 尝试恢复而非直接抛异常。

## 12. 总结

### 核心要点

1. **SafeJSONSession**：异步 I/O + Windows 文件名转义 + JSON 损坏恢复
2. **Workspace 隔离**：每个 Agent 有独立的 Runner、ChannelManager、MemoryManager
3. **ServiceManager 优先级启动**：同优先级并发，不同优先级串行
4. **可复用组件**：热重载时共享 memory_manager，实现记忆保持
5. **final 停止参数**：`True` 停止全部，`False` 保留可复用组件

### 13.4 添加日志的最佳实践

```python
import logging
from qwenpaw.app.runner.session import SafeJSONSession

logger = logging.getLogger(__name__)

class SafeJSONSession:
    def save(self) -> None:
        try:
            with open(self._path, "w", encoding="utf-8") as f:
                json.dump(self._data, f, ensure_ascii=False, indent=2)
            logger.debug(f"Session saved: {self._session_id}")
        except Exception as e:
            logger.error(f"Failed to save session {self._session_id}: {e}")
            raise
```

### 13.5 如何避免破坏架构

**会话管理规则**：
- SafeJSONSession 使用文件锁防止并发写入损坏数据
- Workspace 必须设置适当权限，防止未授权访问
- 服务注册必须在 Agent 初始化前完成

**工作目录规则**：
- 每个会话使用独立工作目录，避免文件冲突
- Workspace 清理必须删除所有创建的文件和目录
- 路径操作使用 `pathlib.Path` 而非字符串拼接

**测试要求**：
- 修改会话管理后，必须测试并发读写场景
- 修改 Workspace 后，必须测试多会话并行场景
- 修改服务管理后，必须测试服务注册和生命周期
