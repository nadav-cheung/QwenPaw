```python
def _discover_custom_channels() -> dict[str, type[BaseChannel]]:
    """从 CUSTOM_CHANNELS_DIR 加载自定义渠道"""
    out: dict[str, type[BaseChannel]] = {}
    if not CUSTOM_CHANNELS_DIR.is_dir():
        return out

    # 将目录加入 sys.path
    dir_str = str(CUSTOM_CHANNELS_DIR)
    if dir_str not in sys.path:
        sys.path.insert(0, dir_str)

    for path in sorted(CUSTOM_CHANNELS_DIR.iterdir()):
        # .py 文件 或 带 __init__.py 的目录
        if path.suffix == ".py" and path.stem != "__init__":
            name = path.stem
        elif path.is_dir() and (path / "__init__.py").exists():
            name = path.name
        else:
            continue

        # 导入模块并查找 BaseChannel 子类
        mod = importlib.import_module(name)
        for obj in vars(mod).values():
            if isinstance(obj, type) and issubclass(obj, BaseChannel) and obj is not BaseChannel:
                key = getattr(obj, "channel", None)
                if key:
                    out[key] = obj
    return out
```

**`is_control_command` 匹配规则** (约 100 行):

```python
def is_control_command(self, query: str) -> bool:
    """检查查询是否匹配注册的命令前缀。

    匹配规则：
    - 必须以 / 开头
    - 前缀完全匹配（前缀后的下一个字符必须是空格/制表符/结尾）
    """
    query_lower = query.strip().lower()
    if not query_lower.startswith("/"):
        return False

    # 按前缀长度降序排列，最长优先匹配
    for prefix in sorted(self._command_to_level.keys(), key=len, reverse=True):
        if query_lower.startswith(prefix):
            next_char_idx = len(prefix)
            if next_char_idx >= len(query_lower):
                return True
            next_char = query_lower[next_char_idx]
            if next_char in (" ", "\t", "\n"):
                return True
    return False
```

### 1.5 ChannelManager 生命周期管理

`ChannelManager`（`channels/manager.py`）统一管理所有渠道实例：

```python
class ChannelManager:
    def __init__(self, channels: List[BaseChannel]):
        self.channels = channels
        self._queue_manager: UnifiedQueueManager | None = None
        self._command_registry = CommandRegistry()
        self._restart_locks: dict[str, asyncio.Lock] = {}  # per-channel restart lock
        self._enqueue_tasks: set[asyncio.Task] = set()      # graceful shutdown

    async def start_all(self) -> None:
        # 1. 初始化 UnifiedQueueManager
        self._queue_manager = UnifiedQueueManager(consumer_fn=self._consume_queue)
        self._queue_manager.start_cleanup_loop()

        # 2. 为每个渠道设置入队回调
        for ch in self.channels:
            ch.set_enqueue(self._make_enqueue_cb(ch.channel))

        # 3. 启动每个渠道
        for g in self.channels:
            await g.start()

    async def stop_all(self) -> None:
        # 1. 取消所有待处理的入队任务
        for task in self._enqueue_tasks:
            task.cancel()
        await asyncio.wait(self._enqueue_tasks, timeout=2.0)

        # 2. 停止队列管理器（停止所有消费者）
        if self._queue_manager is not None:
            await self._queue_manager.stop_all()

        # 3. 停止所有渠道
        for ch in reversed(self.channels):
            await ch.stop()
```

```python
    """Restart a single channel by stopping and re-starting it.

    使用 per-channel lock 防止并发重启导致资源泄漏。

        new_channel = channel_instance.clone(channel_cfg)
```

```python
def _extract_session_id(
    self,
    ch: BaseChannel,
    payload: Any,
) -> str:
    """Extract normalized session_id from payload.

    优先使用 payload 中已有的 session_id，否则委托给 channel 的 get_debounce_key。
    """
    # 检查 payload 是否已有规范化 session_id
    if isinstance(payload, dict):
        existing_sid = payload.get("session_id")
        if existing_sid:
            return existing_sid

    if hasattr(payload, "session_id"):
        existing_sid = payload.session_id
        if existing_sid:
            return existing_sid

    # 委托给 channel 的 get_debounce_key（内部调用 resolve_session_id）
    return ch.get_debounce_key(payload)
```

**ChannelManager 的设计亮点**：
- **per-channel 锁**：每个渠道独立锁，避免全局竞争，类似 Java 的 `ConcurrentHashMap.compute()` 分段锁
- **动态队列创建**：队列按需创建而非预建线程池，类似 Java 的 `LinkedBlockingQueue` 按需创建
- **Idle 清理**：600秒无活动后自动销毁队列，类似 Java 的 `ScheduledExecutorService` 定期清理
- **批量合并**：快速连续的语音/图片消息合并处理，类似 Java 的消息批次聚合

**核心特性**：
1. **动态消费者创建**：队列按需创建，无固定 worker 池
2. **per-session + per-priority 隔离**：不同会话/优先级互不阻塞
3. **自动 idle 队列清理**：600秒无活动后自动销毁
4. **有界队列**：避免内存无限增长（默认 maxsize=1000）

### 2.1 三元组队列键（QueueKey）

消息队列使用 `(channel_id, session_id, priority_level)` 三元组作为键：

```python
# channels/unified_queue_manager.py
QueueKey = Tuple[str, str, int]  # (channel_id, session_id, priority_level)
```

这确保了：
- **跨会话并发**：不同用户的会话互不阻塞
- **同会话按优先级串行**：同一用户的不同优先级消息有序处理
- **动态队列创建**：按需创建，无固定 worker 池限制

```python
async def _consume_queue(self, queue, channel_id, session_id, priority_level):
    """Consumer function for UnifiedQueueManager.

    实现 per-queue 消费者循环，带批量合并逻辑。
    """
    while True:
        # 获取第一个 payload
        payload = await queue.get()

        # 重新获取 channel 实例（每次迭代检查 replace_channel 交换）
        ch = await self.get_channel(channel_id)
        if not ch:
            for _retry in range(3):
                await asyncio.sleep(0.5)
                ch = await self.get_channel(channel_id)
                if ch:
                    break
            if not ch:
                return  # 静默丢弃

        # 批量拉取同队列消息（快速连续的语音/图片合并）
        batch = [payload]
        while True:
            try:
                next_payload = queue.get_nowait()
                batch.append(next_payload)
            except asyncio.QueueEmpty:
                break

        # 处理批量
        await _process_batch(ch, batch)

        # 更新已处理计数
        await self._queue_manager.increment_processed(
            channel_id, session_id, priority_level, count=len(batch)
        )
```

**无文本内容防抖** (`base.py` 约 280 行) - 语音消息特殊处理:

```python
def _apply_no_text_debounce(self, session_id, content_parts):
    """If content has no text, buffer and return (False, []).
    If has text, return (True, merged) with any buffered content prepended.
    Audio-only messages bypass debounce and are processed immediately."""

    if not self._content_has_text(content_parts):
        if self._content_has_audio(content_parts):
            # 语音消息立即处理（完整用户输入）
            pending = self._pending_content_by_session.pop(session_id, [])
            return True, pending + list(content_parts)

        # 缓冲无文本内容
        self._pending_content_by_session.setdefault(session_id, []).extend(content_parts)
        return False, []

    # 有文本，合并缓冲内容
    pending = self._pending_content_by_session.pop(session_id, [])
    return True, pending + list(content_parts)
```

```python
async def _send_via_webhook(self, webhook: str, text: str) -> None:
    """通过会话 Webhook 主动发送消息（push 模式）"""
    import aiohttp
    payload = {
        "msgtype": "markdown",
        "markdown": {"title": "QwenPaw", "text": text},
    }
    async with aiohttp.ClientSession() as session:
        async with session.post(webhook, json=payload) as resp:
            if resp.status != 200:
                raise Exception(f"Webhook push failed: {resp.status}")
```

**热重载流程** (`replace_channel()`):

```
1. 设置入队回调

         │
2. 在锁外启动新渠道（可能耗时）

               break
       if old_channel is not None:
```

```python
        new_channel = channel_instance.clone(channel_cfg)
```

### 8.1 WebSocket vs SSE 差异分析

| 特性 | WebSocket | SSE (Server-Sent Events) |
|------|-----------|--------------------------|
| **方向** | 全双工双向通信 | 单向服务器推送 |
| **连接** | 持久连接 | 持久连接 |
| **协议** | `ws://` / `wss://` | HTTP 长轮询 |
| **复杂度** | 较高（需处理 ping/pong、握手） | 较低（基于 HTTP） |
| **适用场景** | 实时交互、低延迟、频繁双向 | 事件流、推送通知 |
| **重连** | 需手动实现 | 自动重连 |
| **浏览器支持** | 全支持 | 全支持（IE 除外） |

**QwenPaw 中的 SSE 使用场景** (routers/skills_stream.py:170-248):
- AI 技能优化流式响应
- 数据格式: `data: {"text": "增量文本"}\n\n`

**QwenPaw 中的 WebSocket 使用场景**:

```python
class _StreamTracker:
    """追踪流式更新，生成增量 delta"""

    def __init__(self):
        self._snapshots: dict[str, Any] = {}

    def track(self, session_id: str, update: dict) -> dict:
        """将完整快照转换为增量更新"""
        prev = self._snapshots.get(session_id, {})
```

### 10.2 Allowlist 访问控制

**源码路径**: `channels/base.py` 第271-291行

```python
def _check_allowlist(
    self,
    sender_id: str,
    is_group: bool,
) -> tuple[bool, Optional[str]]:
    """检查发送者是否在白名单策略下允许访问"""
    policy = self.group_policy if is_group else self.dm_policy

    # open 策略：允许所有
    if policy == "open":
        return True, None

    # 检查是否在白名单中
    if sender_id in self.allow_from:
        return True, None

    # 返回拒绝消息
    if self.deny_message:
        return False, self.deny_message
    if is_group:
        return False, "Sorry, this bot is only available to authorized users."
    return False, f"Sorry, you are not authorized... Your ID: {sender_id}"
```

| `allowlist` | 仅白名单用户 | `allow_from` 列表 |

## 11. 总结

QwenPaw 的消息渠道系统采用了清晰的分层架构：

| 层次 | 职责 | 核心类 |
|------|------|--------|

| 安全层 | 访问控制、限流 | `ToolGuard`, `SecretStore` |

理解这些层次后，你可以：
- 快速定位问题（通过日志区分层次）
- 扩展新渠道（遵循 BaseChannel 接口）
- 深度定制现有渠道（重写特定方法）
- 集成外部智能体（通过 ACP 协议）
- 配置安全策略（白名单、限流、加密）

### 8.4 如何避免破坏架构

**渠道注册规则**：
- 新渠道必须继承 `BaseChannel` 并实现核心方法
- 渠道名称必须唯一，避免与内置渠道冲突
- 渠道配置必须支持热重载

**消息处理规则**：
- 消息处理必须在异步上下文中进行
- 消息发送失败必须记录日志并重试
- 不要在消息处理中执行耗时的同步操作

**并发安全规则**：
- 使用 `per_channel_lock` 保护共享状态
- Session 状态修改必须使用原子操作
- 队列操作必须考虑竞态条件

**测试要求**：
- 修改渠道管理器后，必须运行 `pytest tests/app/channels/ -v`
- 添加新渠道必须实现完整的单元测试和集成测试
- 修改消息格式后必须验证向后兼容性
