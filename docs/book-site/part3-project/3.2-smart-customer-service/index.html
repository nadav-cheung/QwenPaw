    项目目标¶

    创建一个智能客服系统，实现： - 多渠道统一接入 - 意图识别和 FAQ 匹配 - 转人工和工单系统

    技术架构¶

    多渠道(Channel) → 消息路由 → 意图识别(Agent) → FAQ/Memory → 响应

    1.1 FAQ 数据结构¶

    文件: faq_kb.py

    """FAQ 知识库"""

from dataclasses import dataclass
from typing import Optional

@dataclass
class FAQ:
    """FAQ 条目"""
    id: str
    question: str
    answer: str
    keywords: list[str]
    category: str
    confidence_threshold: float = 0.7

class FAQKnowledgeBase:
    """FAQ 知识库"""

    def __init__(self):
        self.faqs: list[FAQ] = []

    def add(self, faq: FAQ):
        self.faqs.append(faq)

    def search(self, query: str) -> list[tuple[FAQ, float]]:
        """搜索最匹配的 FAQ，返回 (faq, score) 列表"""
        scores = []
        query_lower = query.lower()

        for faq in self.faqs:
            # 简单关键词匹配
            score = sum(1 for kw in faq.keywords if kw.lower() in query_lower)
            score += sum(0.5 for kw in faq.keywords if kw.lower() in faq.question.lower())

            if score > 0:
                scores.append((faq, score / len(faq.keywords)))

        # 按分数排序
        scores.sort(key=lambda x: x[1], reverse=True)
        return scores

    def load_from_file(self, filepath: str):
        """从文件加载 FAQ"""
        import json
        with open(filepath, "r", encoding="utf-8") as f:
            data = json.load(f)
            for item in data:
                self.add(FAQ(**item))

    2.2 客服智能体代码¶

    文件: customer_service.py

    """智能客服系统"""

from dataclasses import dataclass
from typing import Optional
from datetime import datetime
from qwenpaw import QwenPawAgent
from faq_kb import FAQKnowledgeBase, FAQ

@dataclass
class CustomerQuery:
    """用户查询"""
    content: str
    channel: str
    user_id: str
    timestamp: str

class CustomerServiceAgent:
    """客服智能体"""

    def __init__(self, config_path: str, faq_file: str):
        self.agent = QwenPawAgent(config=config_path)
        self.faq_kb = FAQKnowledgeBase()
        self.faq_kb.load_from_file(faq_file)

        # 转人工阈值
        self.human_threshold = 0.3

        # 意图分类提示
        self.intent_prompt = """请判断用户意图，只回答一个词：
- faq: 常见问题
- human: 需要人工客服
- other: 其他

用户消息：{message}
意图："""

    async def classify_intent(self, message: str) -> str:
        """意图分类"""
        response = await self.agent.run(
            self.intent_prompt.format(message=message)
        )
        response = response.strip().lower()

        if "faq" in response:
            return "faq"
        elif "human" in response:
            return "human"
        return "other"

    async def find_faq(self, message: str) -> Optional[tuple[FAQ, float]]:
        """查找匹配的 FAQ"""
        matches = self.faq_kb.search(message)

        if not matches:
            return None

        best_faq, best_score = matches[0]

        # 如果置信度低于阈值，转人工
        if best_score < self.human_threshold:
            return None

        return (best_faq, best_score)

    async def handle_query(self, query: CustomerQuery) -> str:
        """处理用户查询"""
        # 1. 意图分类
        intent = await self.classify_intent(query.content)

        if intent == "faq":
            # 2. 查找 FAQ
            result = await self.find_faq(query.content)

            if result:
                faq, score = result
                return f"{faq.answer}\n\n💡 这是 {faq.category} 类问题吗？"

            # FAQ 置信度不够，转人工
            return await self.transfer_to_human(query, reason="faq_low_confidence")

        elif intent == "human":
            return await self.transfer_to_human(query, reason="user_requested")

        # 3. 其他问题，用 AI 回答
        response = await self.agent.run(query.content)

        # 添加建议
        response += "\n\n❓ 还有其他问题吗？回复「人工」转接客服。"

        return response

    async def transfer_to_human(self, query: CustomerQuery, reason: str) -> str:
        """转人工客服"""
        # 发送通知给客服
        notification = f"""🔔 新客服请求
渠道：{query.channel}
用户：{query.user_id}
原因：{reason}
问题：{query.content}"""

        await self.agent.send_to_channel("internal", notification)

        return "请稍等，正在为您转接人工客服..."

    async def process(self, message: str, channel: str, user_id: str) -> str:
        """处理消息"""
        query = CustomerQuery(
            content=message,
            channel=channel,
            user_id=user_id,
            timestamp=datetime.now().isoformat()
        )

        return await self.handle_query(query)

    文件: channel_adapter.py

    """渠道适配器"""

from abc import ABC, abstractmethod
from typing import Any

class ChannelAdapter(ABC):
    """渠道适配器基类"""

    @abstractmethod
    def extract_text(self, raw_message: Any) -> str:
        """从渠道消息中提取文本"""
        pass

    @abstractmethod
    def format_response(self, text: str) -> Any:
        """格式化响应"""
        pass

    @abstractmethod
    def extract_user_id(self, raw_message: Any) -> str:
        """从渠道消息中提取用户 ID"""
        pass

class DingTalkAdapter(ChannelAdapter):
    """钉钉适配器"""

    def extract_text(self, raw_message: dict) -> str:
        # 钉钉消息格式
        return raw_message.get("text", {}).get("content", "")

    def format_response(self, text: str) -> dict:
        return {
            "msgtype": "text",
            "text": {"content": text}
        }

    def extract_user_id(self, raw_message: dict) -> str:
        return raw_message.get("sender_nick", "unknown")

class FeishuAdapter(ChannelAdapter):
    """飞书适配器"""

    def extract_text(self, raw_message: dict) -> str:
        # 飞书消息格式
        return raw_message.get("text", {}).get("content", "")

    def format_response(self, text: str) -> dict:
        return {
            "msg_type": "text",
            "content": {"text": text}
        }

    def extract_user_id(self, raw_message: dict) -> str:
        return raw_message.get("sender", {}).get("sender_id", {}).get("open_id", "unknown")

class DiscordAdapter(ChannelAdapter):
    """Discord 适配器"""

    def extract_text(self, raw_message: dict) -> str:
        # Discord 消息格式
        return raw_message.get("content", "")

    def format_response(self, text: str) -> dict:
        return {"content": text}

    def extract_user_id(self, raw_message: dict) -> str:
        return raw_message.get("author", {}).get("id", "unknown")

import asyncio
from datetime import datetime
from customer_service import CustomerServiceAgent
from channel_adapter import DingTalkAdapter, FeishuAdapter, DiscordAdapter

# 初始化
agent = CustomerServiceAgent(
    config_path="config/customer_service.yaml",
    faq_file="data/faqs.json"
)

# 适配器映射
adapters = {
    "dingtalk": DingTalkAdapter(),
    "feishu": FeishuAdapter(),
    "discord": DiscordAdapter(),
}

async def handle_dingtalk_webhook(request: dict):
    """处理钉钉 webhook 请求"""
    adapter = adapters["dingtalk"]

    text = adapter.extract_text(request)
    user_id = adapter.extract_user_id(request)

    response_text = await agent.process(
        message=text,
        channel="dingtalk",
        user_id=user_id
    )

    return adapter.format_response(response_text)

async def handle_feishu_webhook(request: dict):
    """处理飞书 webhook 请求"""
    adapter = adapters["feishu"]

    text = adapter.extract_text(request)
    user_id = adapter.extract_user_id(request)

    response_text = await agent.process(
        message=text,
        channel="feishu",
        user_id=user_id
    )

    return adapter.format_response(response_text)

async def handle_discord_message(message: dict):
    """处理 Discord 消息"""
    adapter = adapters["discord"]

    text = adapter.extract_text(message)
    user_id = adapter.extract_user_id(message)

    response_text = await agent.process(
        message=text,
        channel="discord",
        user_id=user_id
    )

    return adapter.format_response(response_text)

if __name__ == "__main__":

    import unittest
from faq_kb import FAQKnowledgeBase, FAQ

class TestFAQKnowledgeBase(unittest.TestCase):
    def setUp(self):
        self.kb = FAQKnowledgeBase()
        self.kb.add(FAQ(
            id="test_001",
            question="如何重置密码",
            answer="访问 reset 页",
            keywords=["密码", "重置"],
            category="账户"
        ))

    def test_search_exact_keyword(self):
        results = self.kb.search("密码")
        self.assertGreater(len(results), 0)

    def test_search_partial_match(self):
        results = self.kb.search("我想重置")

    def test_search_no_match(self):
        results = self.kb.search("完全不相关的内容 xyz")

        results = self.kb.search("")
        self.assertEqual(len(results), 0)

class FeedbackManager:

        # 如果差评，通知人工

    async def notify_human(self, message: str):
        """通知人工客服"""
        print(f"通知人工客服: {message}")

    async def handle_followup(self, query: CustomerQuery, context: dict) -> str:
    """处理多轮对话"""
    # 获取对话历史
    history = await self.agent.memory.get_recent(max_turns=5)

    # 追加当前问题
    history.append({"role": "user", "content": query.content})

    # 带上下文回答
    response = await self.agent.run_with_history(history)

    return response

async def handle_conversation_flow(self, query: CustomerQuery) -> str:
    """处理对话流程"""
    # 检查是否在对话中
    session_key = f"{query.channel}:{query.user_id}"

    if session_key in self.active_sessions:
        session = self.active_sessions[session_key]

        # 如果用户确认/否定
        if query.content in ["是", "是的", "对"]:
            session.acknowledged = True
            return "感谢您的确认，如有其他问题随时联系我们。"

        if query.content in ["否", "不是", "不对"]:
            session.acknowledged = False
            return await self.transfer_to_human(query, reason="user_correction")

    # 新对话
    return await self.handle_query(query)

    扩展 3：工单系统¶

    @dataclass
class Ticket:
    """工单"""
    id: str
    user_id: str
    channel: str
    question: str
    status: str  # open, pending, resolved, closed
    assigned_to: Optional[str]
    created_at: datetime
    updated_at: datetime

class TicketSystem:
    """工单系统"""

    def __init__(self):
        self.tickets: dict[str, Ticket] = {}

    def generate_id(self) -> str:
        """生成工单 ID"""
        import uuid
        return f"TKT-{uuid.uuid4().hex[:8].upper()}"

    async def create_ticket(self, query: CustomerQuery, reason: str) -> Ticket:
        """创建工单"""
        ticket = Ticket(
            id=self.generate_id(),
            user_id=query.user_id,
            channel=query.channel,
            question=query.content,
            status="open",
            assigned_to=None,
            created_at=datetime.now(),
            updated_at=datetime.now()
        )

        self.tickets[ticket.id] = ticket

        # 发送创建通知
        await self.notify_ticket_created(ticket)

        return ticket

    async def assign_ticket(self, ticket_id: str, agent_id: str):
        """分配工单"""
        if ticket_id in self.tickets:
            self.tickets[ticket_id].assigned_to = agent_id
            self.tickets[ticket_id].status = "pending"
            self.tickets[ticket_id].updated_at = datetime.now()

    async def resolve_ticket(self, ticket_id: str):
        """解决工单"""
        if ticket_id in self.tickets:
            self.tickets[ticket_id].status = "resolved"
            self.tickets[ticket_id].updated_at = datetime.now()

    async def notify_ticket_created(self, ticket: Ticket):
        """通知工单创建"""
        message = f"""📝 新工单创建
工单号: {ticket.id}
用户: {ticket.user_id}
渠道: {ticket.channel}
问题: {ticket.question}"""
        print(message)
