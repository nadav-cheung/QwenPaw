    1. Channel 架构¶

    1.1 抽象基类¶

    源码路径: src/qwenpaw/app/channels/base.py

    BaseChannel 是所有渠道的抽象基类，定义了渠道的通用接口：

    class BaseChannel(ABC):
    """渠道抽象基类"""

    channel: ChannelType  # 渠道标识符，如 "dingtalk", "telegram"

    # 是否使用管理器队列
    uses_manager_queue: bool = True

    @classmethod
    def from_env(cls, process, on_reply_sent=None) -> "BaseChannel":
        """从环境变量创建渠道"""

    @classmethod
    def from_config(cls, process, config, on_reply_sent=None, ...) -> "BaseChannel":
        """从配置创建渠道"""

    async def start(self) -> None:
        """启动渠道"""

    async def stop(self) -> None:
        """停止渠道"""

    async def send(self, to_handle: str, text: str, meta: Optional[Dict] = None) -> None:
        """发送消息"""

    关键内部属性 (用于消息处理流程):

    属性 类型 说明
    _process ProcessHandler AgentRequest → AsyncIterator[Event] 的处理函数
    _enqueue EnqueueCallback ChannelManager 设置的入队回调
    _workspace Any Workspace 实例，用于 TaskTracker 集成
    _debounce_seconds float 时间防抖间隔（默认 0.0，子类可覆盖如 0.3）
    _debounce_pending Dict[str, List] 时间防抖待处理 payload
    _pending_content_by_session Dict[str, List] 无文本内容缓冲（等待语音识别结果）

    2. ChannelManager¶

    2.1 核心类¶

    源码路径: src/qwenpaw/app/channels/manager.py

    ChannelManager 统一管理所有渠道实例的生命周期：

    class ChannelManager:
    """管理所有渠道实例"""

    def __init__(self, channels: List[BaseChannel]):
        self.channels = channels
        self._lock = asyncio.Lock()
        self._loop: Optional[asyncio.AbstractEventLoop] = None
        self._queue_manager: UnifiedQueueManager | None = None
        self._command_registry = CommandRegistry()

        # Per-channel locks to prevent concurrent restarts
        self._restart_locks: dict[str, asyncio.Lock] = {}

        # Track enqueue tasks for graceful shutdown
        self._enqueue_tasks: set[asyncio.Task] = set()

    核心方法:

    方法 功能
    from_env() 从环境变量创建渠道
    from_config() 从配置文件创建渠道
    start_all() 启动所有渠道和队列管理器
    stop_all() 停止所有渠道和队列管理器
    enqueue() 线程安全入队（WebSocket/轮询线程调用）
    get_channel() 获取指定渠道实例
    restart_channel() 热重载单个渠道
    replace_channel() 替换渠道实例

    2.2 生命周期¶

    创建 → 注册 → 启动 → 运行 → 停止 → 清理

    启动流程 (start_all):

    async def start_all(self) -> None:
    self._loop = asyncio.get_running_loop()

    # 1. 初始化 UnifiedQueueManager
    self._queue_manager = UnifiedQueueManager(
        consumer_fn=self._consume_queue,
        queue_maxsize=_CHANNEL_QUEUE_MAXSIZE,
    )
    self._queue_manager.start_cleanup_loop()

    # 2. 为每个渠道设置入队回调
    for ch in self.channels:
        if getattr(ch, "uses_manager_queue", True):
            ch.set_enqueue(self._make_enqueue_cb(ch.channel))

    # 3. 启动每个渠道
    for g in self.channels:
        await g.start()

    停止流程 (stop_all):

    async def stop_all(self) -> None:
    # 1. 取消所有待处理的入队任务
    for task in self._enqueue_tasks:
        task.cancel()

    # 2. 停止队列管理器（停止所有消费者）
    if self._queue_manager is not None:
        await self._queue_manager.stop_all()

    # 3. 停止所有渠道
    for ch in reversed(self.channels):
        await ch.stop()

    3. Webhook 实现¶

    3.1 Webhook 接收¶

    Webhook 渠道通过 HTTP POST 接收外部平台的回调请求。以 DingTalk 为例：

    源码路径: src/qwenpaw/app/channels/dingtalk/channel.py

    class DingTalkChannel(BaseChannel):
    """钉钉机器人渠道"""

    async def handle_webhook(self, request: Request) -> Response:
        """处理传入的 Webhook 请求"""
        payload = await request.json()
        # 验证签名
        if not self._verify_signature(payload):
            return Response(status_code=401)
        # 转换消息格式
        msg = self._parse_dingtalk_payload(payload)
        # 入队处理
        self._enqueue(msg)
        return Response(json={"status": "ok"})

    3.2 安全验证¶

    Webhook 请求需要验证签名以确保来源可信：

    def _verify_signature(self, payload: dict, signature: str) -> bool:
    """验证 Webhook 签名

    Args:
        payload: 请求体
        signature: 请求中的签名

    Returns:
        签名是否有效
    """
    timestamp = payload.get("timestamp", "")
    secret = self._secret

    # 钉钉签名算法：HMAC-SHA256
    import hmac
    import hashlib
    import base64

    string_to_sign = f"{timestamp}\n{secret}"
    encoded = base64.b64encode(
        hmac.new(
            secret.encode(),
            string_to_sign.encode(),
            hashlib.sha256
        ).digest()
    ).decode()

    return encoded == signature

    4. 消息队列¶

    4.1 队列机制¶

    源码路径: src/qwenpaw/app/channels/unified_queue_manager.py

    UnifiedQueueManager 实现基于三元组键的动态队列系统：

    QueueKey = Tuple[str, str, int]  # (channel_id, session_id, priority_level)

class UnifiedQueueManager:
    """统一队列管理器"""

    def __init__(
        self,
        consumer_fn: ConsumerFn,
        queue_maxsize: int = 1000,
        idle_timeout: float = 600.0,   # 10分钟 idle 后清理
        cleanup_interval: float = 60.0,  # 每分钟检查一次
    ):
        self._queues: Dict[QueueKey, QueueState] = {}

    QueueState 数据结构:

    @dataclass
class QueueState:
    """单个队列的状态"""
    queue: asyncio.Queue
    consumer_task: asyncio.Task[None]
    created_at: float = field(default_factory=time.time)
    last_activity: float = field(default_factory=time.time)
    processed_count: int = 0

    设计特点:

    特性 说明
    动态创建 队列按需创建，而非预建线程池
    会话隔离 不同 session_id 的消息并发处理
    优先级队列 相同会话内按优先级顺序处理
    空闲清理 600秒无活动后自动销毁队列

    4.2 防抖机制¶

    源码路径: src/qwenpaw/app/channels/base.py

    防抖机制用于合并快速连续的消息（如图片批量发送）：

    async def consume_one(self, payload: Any) -> None:
    """处理单个 payload，带防抖"""
    if self._debounce_seconds > 0 and self._is_native_payload(payload):
        key = self.get_debounce_key(payload)

        # 合并到待处理队列
        self._debounce_pending.setdefault(key, []).append(payload)

        # 重置防抖定时器
        old = self._debounce_timers.pop(key, None)
        if old and not old.done():
            old.cancel()

        async def flush(k: str) -> None:
            await asyncio.sleep(self._debounce_seconds)
            items = self._debounce_pending.pop(k, [])
            if not items:
                return
            merged = self.merge_native_items(items)
            await self._consume_one_request(merged)

        self._debounce_timers[key] = asyncio.create_task(flush(key))
        return

    await self._consume_one_request(payload)

    防抖类型:

    类型 说明 场景
    时间防抖 _debounce_seconds 间隔内的消息合并 快速连续输入
    内容防抖 无文本内容的消息缓冲（等待语音识别） 语音消息

    5. 命令优先级路由¶

    5.1 CommandRegistry¶

    源码路径: src/qwenpaw/app/channels/command_registry.py

    CommandRegistry 管理命令到优先级的映射：

    class CommandRegistry:
    def __init__(self):
        self._priority_names: Dict[str, int] = {
            "critical": 0,   # 紧急控制命令
            "high": 10,      # 高优先级查询
            "normal": 20,    # 普通消息（默认）
            "low": 30,       # 低优先级批处理
        }

    def _register_defaults(self) -> None:
        # Critical (0): /stop
        self.register_command("/stop", priority_level=0)

        # High (10): /daemon 系列 + /status 等
        self.register_command("/daemon status", priority_level=10)
        self.register_command("/status", priority_level=10)

    优先级路由流程:

    消息入队 → 提取查询文本 → CommandRegistry.get_priority_level()
         → UnifiedQueueManager.enqueue(channel_id, session_id, priority_level, payload)

    5.2 控制命令检测¶

    def is_control_command(self, query: str) -> bool:
    """检查查询是否匹配注册的命令前缀"""
    query_lower = query.strip().lower()
    if not query_lower.startswith("/"):
        return False

    # 最长前缀优先匹配
    for prefix in sorted(self._command_to_level.keys(), key=len, reverse=True):
        if query_lower.startswith(prefix):
            next_char_idx = len(prefix)
            if next_char_idx >= len(query_lower):
                return True
            next_char = query_lower[next_char_idx]
            if next_char in (" ", "\t", "\n"):
                return True
    return False

    7. 设计意图¶

    7.1 为什么要消息队列¶

    消息队列解耦了消息接收和处理：

    优势 说明
    缓冲 吸收流量峰值，防止系统过载
    隔离 接收线程不阻塞处理逻辑
    批量 合并快速连续的消息，减少 API 调用
    优先级 重要消息（如控制命令）优先处理

    7.2 权衡取舍¶

    设计选择 权衡
    动态队列 vs 固定线程池 动态队列节省资源，但创建有开销
    内存队列 vs 外部队列 内存队列低延迟，但重启丢消息
    600s 空闲清理 平衡资源回收和队列重建成本
