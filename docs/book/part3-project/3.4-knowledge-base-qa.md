## 项目概述

### 项目目标
创建一个企业知识库问答系统，实现：

```
用户问题 → 向量检索 → 上下文组装 → LLM 生成 → 回答

from dataclasses import dataclass
from datetime import datetime
from typing import Optional
import numpy as np

@dataclass
class Document:
    """文档"""
    id: str
    content: str
    metadata: dict  # title, source, created_at, etc.
    embedding: Optional[np.ndarray] = None
    
    def __post_init__(self):
        if isinstance(self.embedding, list):
            self.embedding = np.array(self.embedding)

@dataclass
class Chunk:
    """文档块"""
    id: str
    document_id: str
    content: str
    chunk_index: int  # 在文档中的位置
    metadata: dict
    embedding: Optional[np.ndarray] = None

@dataclass
class RetrievalResult:
    """检索结果"""
    chunk: Chunk
    score: float  # 相似度分数
    distance: float  # 向量距离

"""向量化器"""

from abc import ABC, abstractmethod
from typing import list
import numpy as np

class Embedder(ABC):
    """向量化器抽象基类"""
    
    @abstractmethod
    async def embed(self, texts: list[str]) -> list[np.ndarray]:
        """将文本转为向量"""
        pass
    
    @abstractmethod
    def get_dimension(self) -> int:
        """获取向量维度"""
        pass

class OpenAIEmbedder(Embedder):
    """OpenAI 向量化"""
    
    def __init__(self, api_key: str, model: str = "text-embedding-3-small"):
        self.client = OpenAI(api_key=api_key)
        self.model = model
        self.dimension = 1536  # text-embedding-3-small
    
    async def embed(self, texts: list[str]) -> list[np.ndarray]:
        response = await self.client.embeddings.create(
            model=self.model,
            input=texts
        )
        
        return [
            np.array(item.embedding) 
            for item in response.data
        ]
    
    def get_dimension(self) -> int:
        return self.dimension

class OllamaEmbedder(Embedder):
    """Ollama 本地向量化"""
    
    def __init__(self, base_url: str = "http://localhost:11434", model: str = "nomic-embed-text"):
        self.base_url = base_url
        self.model = model
        self.dimension = 768  # nomic-embed-text
    
    async def embed(self, texts: list[str]) -> list[np.ndarray]:
        import aiohttp
        
        embeddings = []
        async with aiohttp.ClientSession() as session:
            for text in texts:
                async with session.post(
                    f"{self.base_url}/api/embeddings",
                    json={"model": self.model, "prompt": text}
                ) as resp:
                    data = await resp.json()
                    embeddings.append(np.array(data["embedding"]))
        
        return embeddings
    
    def get_dimension(self) -> int:
        return self.dimension
```

### 步骤 2：向量存储

**文件**: `kb/vector_store.py`

```python
"""向量存储"""

from abc import ABC, abstractmethod
from typing import list
import numpy as np

class VectorStore(ABC):
    """向量存储抽象基类"""
    
    @abstractmethod
    async def add(self, chunks: list[Chunk]):
        """添加文档块"""
        pass
    
    @abstractmethod
    async def search(self, query_embedding: np.ndarray, top_k: int) -> list[RetrievalResult]:
        """搜索最相似的文档块"""
        pass
    
    @abstractmethod
    async def delete(self, chunk_ids: list[str]):
        """删除文档块"""
        pass

class ChromaStore(VectorStore):
    """ChromaDB 向量存储"""
    
    def __init__(self, collection_name: str, embedder: Embedder):
        import chromadb
        self.client = chromadb.Client()
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}  # 余弦相似度
        )
        self.embedder = embedder
    
    async def add(self, chunks: list[Chunk]):
        # 向量化
        texts = [chunk.content for chunk in chunks]
        embeddings = await self.embedder.embed(texts)
        
        # 添加到 ChromaDB
        self.collection.add(
            ids=[chunk.id for chunk in chunks],
            embeddings=[e.tolist() for e in embeddings],
            documents=texts,
            metadatas=[chunk.metadata for chunk in chunks]
        )
    
    async def search(self, query_embedding: np.ndarray, top_k: int) -> list[RetrievalResult]:
        results = self.collection.query(
            query_embeddings=[query_embedding.tolist()],
            n_results=top_k
        )
        
        retrieval_results = []
        for i in range(len(results["ids"][0])):
            chunk = Chunk(
                id=results["ids"][0][i],
                document_id=results["metadatas"][0][i].get("document_id", ""),
                content=results["documents"][0][i],
                chunk_index=results["metadatas"][0][i].get("chunk_index", 0),
                metadata=results["metadatas"][0][i]
            )
            
            retrieval_results.append(RetrievalResult(
                chunk=chunk,
                score=1 - results["distances"][0][i],  # 距离转相似度
                distance=results["distances"][0][i]
            ))
        
        return retrieval_results
    
    async def delete(self, chunk_ids: list[str]):
        self.collection.delete(ids=chunk_ids)
```

### 步骤 3：文档处理

**文件**: `kb/document_processor.py`

```python
"""文档处理器"""

import re
from typing import list
from pathlib import Path
from kb.models import Document, Chunk
import uuid

class DocumentProcessor:
    """文档处理器"""
    
    def __init__(self, chunk_size: int = 500, chunk_overlap: int = 50):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
    
    def process_file(self, file_path: str) -> list[Chunk]:
        """处理文件，返回文档块"""
        path = Path(file_path)
        
        if path.suffix == ".txt":
            return self.process_text(path.read_text(encoding="utf-8"), {"source": str(path)})
        elif path.suffix == ".md":
            return self.process_markdown(path.read_text(encoding="utf-8"), {"source": str(path)})
        elif path.suffix == ".pdf":
            return self.process_pdf(file_path)
        else:
            raise ValueError(f"Unsupported file type: {path.suffix}")
    
    def process_text(self, text: str, metadata: dict) -> list[Chunk]:
        """处理纯文本"""
        # 简单的滑动窗口分块
        chunks = []
        
        start = 0
        while start < len(text):
            end = start + self.chunk_size
            chunk_text = text[start:end]
            
            chunks.append(Chunk(
                id=str(uuid.uuid4()),
                document_id=metadata.get("document_id", ""),
                content=chunk_text.strip(),
                chunk_index=len(chunks),
                metadata=metadata
            ))
            
            start += self.chunk_size - self.chunk_overlap
        
        return chunks
    
    def process_markdown(self, text: str, metadata: dict) -> list[Chunk]:
        """处理 Markdown，按标题分段"""
        # 按标题分割
        sections = re.split(r'\n(?=#+\s)', text)
        
        chunks = []
        for i, section in enumerate(sections):
            if len(section.strip()) < 50:  # 跳过太短的段
                continue
            
            chunks.append(Chunk(
                id=str(uuid.uuid4()),
                document_id=metadata.get("document_id", ""),
                content=section.strip(),
                chunk_index=i,
                metadata={**metadata, "type": "section"}
            ))
        
        return chunks
    
    def process_pdf(self, file_path: str) -> list[Chunk]:
        """处理 PDF"""
        import pypdf
        
        chunks = []
        reader = pypdf.PdfReader(file_path)
        
        for page_num, page in enumerate(reader.pages):
            text = page.extract_text()
            
            if len(text.strip()) < 50:
                continue
            
            chunks.append(Chunk(
                id=str(uuid.uuid4()),
                document_id=metadata.get("document_id", ""),
                content=text.strip(),
                chunk_index=page_num,
                metadata={**metadata, "page": page_num + 1}
            ))
        
        return chunks
    
    def create_document(self, content: str, metadata: dict) -> Document:
        """创建文档"""
        return Document(
            id=str(uuid.uuid4()),
            content=content,
            metadata=metadata
        )
```

### 步骤 4：RAG 引擎

**文件**: `rag_engine.py`

```python
"""RAG 引擎"""

from typing import list, Optional
import numpy as np
from qwenpaw import QwenPawAgent
from kb.embeddings import Embedder
from kb.vector_store import VectorStore
from kb.models import RetrievalResult

class RAGEngine:
    """RAG 检索增强生成引擎"""
    
    def __init__(
        self,
        agent: QwenPawAgent,
        embedder: Embedder,
        vector_store: VectorStore
    ):
        self.agent = agent
        self.embedder = embedder
        self.vector_store = vector_store
        
        # RAG 提示模板
        self.prompt_template = """你是一个企业知识库问答助手。请根据以下上下文信息回答用户问题。

如果上下文中没有相关信息，请回答"我没有找到相关的答案"。

上下文信息：
{context}

用户问题：{question}

回答："""
    
    async def retrieve(self, query: str, top_k: int = 5) -> list[RetrievalResult]:
        """检索相关文档"""
        # 向量化查询
        query_embedding = (await self.embedder.embed([query]))[0]
        
        # 搜索
        results = await self.vector_store.search(query_embedding, top_k)
        
        return results
    
    def format_context(self, results: list[RetrievalResult]) -> str:
        """格式化上下文"""
        context_parts = []
        
        for i, result in enumerate(results, 1):
            source = result.chunk.metadata.get("source", "Unknown")
            page = result.chunk.metadata.get("page", "")
            source_info = f"{source} (Page {page})" if page else source
            
            context_parts.append(
                f"[{i}] 来源：{source_info}\n"
                f"内容：{result.chunk.content}\n"
                f"相关度：{result.score:.2f}"
            )
        
        return "\n\n".join(context_parts)
    
    async def generate(self, question: str, context: str) -> str:
        """生成回答"""
        prompt = self.prompt_template.format(
            question=question,
            context=context
        )
        
        response = await self.agent.run(prompt)
        return response
    
    async def query(self, question: str, top_k: int = 5) -> str:
        """完整的 RAG 查询"""
        # 1. 检索
        results = await self.retrieve(question, top_k)
        
        if not results:
            return "我在知识库中没有找到相关信息。"
        
        # 2. 格式化上下文
        context = self.format_context(results)
        
        # 3. 生成回答
        response = await self.generate(question, context)
        
        # 4. 添加参考来源
        response += "\n\n📚 参考来源："
        for i, result in enumerate(results[:3], 1):
            source = result.chunk.metadata.get("source", "Unknown")
            response += f"\n[{i}] {source}"
        
        return response
```

### 步骤 5：知识库管理

**文件**: `kb/manager.py`

```python
"""知识库管理器"""

import asyncio
from pathlib import Path
from typing import list
from kb.document_processor import DocumentProcessor
from kb.vector_store import VectorStore
from kb.models import Document, Chunk

class KnowledgeBaseManager:
    """知识库管理器"""
    
    def __init__(self, vector_store: VectorStore, processor: DocumentProcessor):
        self.vector_store = vector_store
        self.processor = processor
        self.documents: dict[str, Document] = {}
    
    async def add_documents(self, file_paths: list[str]) -> int:
        """批量添加文档"""
        total_chunks = 0
        
        for file_path in file_paths:
            # 创建文档
            content = Path(file_path).read_text(encoding="utf-8")
            doc = self.processor.create_document(content, {
                "source": file_path,
                "filename": Path(file_path).name
            })
            self.documents[doc.id] = doc
            
            # 分块
            chunks = self.processor.process_file(file_path)
            
            # 设置文档 ID
            for chunk in chunks:
                chunk.document_id = doc.id
            
            # 添加到向量存储
            await self.vector_store.add(chunks)
            total_chunks += len(chunks)
        
        return total_chunks
    
    async def remove_document(self, document_id: str) -> bool:
        """删除文档"""
        if document_id not in self.documents:
            return False
        
        # 删除文档及其所有块
        chunks = [c for c in self.documents[document_id] if c.id]
        await self.vector_store.delete(chunks)
        
        del self.documents[document_id]
        return True
    
    def list_documents(self) -> list[dict]:
        """列出所有文档"""
        return [
            {
                "id": doc.id,
                "metadata": doc.metadata
            }
            for doc in self.documents.values()
        ]
```

### 步骤 6：主程序

**文件**: `main.py`

```python
"""企业知识库问答入口"""

import asyncio
from qwenpaw import QwenPawAgent
from kb.embeddings import OpenAIEmbedder
from kb.vector_store import ChromaStore
from kb.document_processor import DocumentProcessor
from kb.manager import KnowledgeBaseManager
from rag_engine import RAGEngine

async def main():
    # 1. 初始化组件
    agent = QwenPawAgent(config="config/qa_agent.yaml")
    embedder = OpenAIEmbedder(api_key="${OPENAI_API_KEY}")
    vector_store = ChromaStore(
        collection_name="company_knowledge",
        embedder=embedder
    )
    processor = DocumentProcessor(chunk_size=500)
    
    # 2. 创建知识库管理器
    kb_manager = KnowledgeBaseManager(vector_store, processor)
    
    # 3. 创建 RAG 引擎
    rag = RAGEngine(agent, embedder, vector_store)
    
    # 4. 添加文档
    print("正在导入文档...")
    chunks = await kb_manager.add_documents([
        "docs/faq.md",
        "docs/products.md",
        "docs/policy.md"
    ])
    print(f"已导入 {chunks} 个文档块")
    
    # 5. 问答循环
    print("\n知识库问答系统已启动！输入 quit 退出。")
    
    while True:
        question = input("\n问题：").strip()
        
        if question.lower() == "quit":
            break
        
        if not question:
            continue
        
        answer = await rag.query(question)
        print(f"\n回答：{answer}")

if __name__ == "__main__":
    asyncio.run(main())
```

        """使用 Cross-Encoder 重新排序"""
