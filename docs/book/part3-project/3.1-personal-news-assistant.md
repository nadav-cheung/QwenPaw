### 3.2 智能体代码

**文件**: `news_assistant.py`

```python
"""个人资讯助手"""

import feedparser
from datetime import datetime
from qwenpaw import QwenPawAgent


class NewsAssistant:
    """个人资讯助手：抓取新闻、生成摘要、推送消息"""

    def __init__(self, config_path: str):
        self.agent = QwenPawAgent(config=config_path)
        self.feed_urls = [
            "https://techcrunch.com/feed/",
            "https://www.theverge.com/rss/index.xml",
        ]

    async def fetch_feeds(self) -> list[dict]:
        """抓取 RSS 源"""
        articles = []
        for url in self.feed_urls:
            feed = feedparser.parse(url)
            for entry in feed.entries[:5]:  # 每个源取最新5篇
                articles.append({
                    "title": entry.title,
                    "link": entry.link,
                    "published": entry.get("published", ""),
                    "summary": entry.get("summary", "")[:200],
                })
        return articles

    async def summarize_article(self, article: dict) -> str:
        """使用 AI 生成摘要"""
        prompt = f"""请为以下文章生成一句话摘要：

标题：{article['title']}
来源：{article['link']}

摘要："""

        response = await self.agent.run(prompt)
        return response

    async def fetch_and_summarize(self) -> str:
        """抓取并汇总"""
        # 1. 抓取
        articles = await self.fetch_feeds()

        # 2. 生成摘要
        summaries = []
        for article in articles:
            summary = await self.summarize_article(article)
            summaries.append(f"• {article['title']}\n  {summary}")

        # 3. 组装消息
        message = f"📰 今日资讯 ({datetime.now().strftime('%Y-%m-%d')})\n\n"
        message += "\n\n".join(summaries)

        # 4. 推送
        await self.agent.send_to_channel("dingtalk", message)

        return f"已推送 {len(articles)} 条资讯"
```

## 4. 配置定时任务

### 4.1 注册定时任务

**文件**: `main.py`

```python
"""个人资讯助手入口"""
import asyncio
from qwenpaw.app.scheduler import CronManager


async def main():
    """主入口"""
    # 启动定时任务
    scheduler = CronManager()

    @scheduler.register("0 8 * * *")  # 每天早上8点
    async def morning_news():
        from news_assistant import NewsAssistant
        assistant = NewsAssistant("config/news_assistant.yaml")
        await assistant.fetch_and_summarize()

    await scheduler.start()

    # 保持运行
    while True:
        await asyncio.sleep(3600)


if __name__ == "__main__":
    asyncio.run(main())
```

### 9.1 关键词过滤

```python
async def should_include(self, article: dict) -> bool:
    """根据关键词过滤"""
    keywords = ["AI", "Python", "OpenAI", "GPT"]
    title = article["title"].lower()
    return any(kw.lower() in title for kw in keywords)
```

### 9.2 分类推送

```python
async def categorize(self, article: dict) -> str:
    """文章分类"""
    categories = {
        "AI": ["AI", "GPT", "Claude", "machine learning"],
        "Tech": ["iPhone", "Android", "Google"],
        "Startup": ["funding", "raises", " Series"],
    }

    for cat, keywords in categories.items():
        if any(kw.lower() in article["title"].lower() for kw in keywords):
            return cat
    return "Other"
```
