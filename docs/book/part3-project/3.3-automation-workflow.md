# 第 3.3 章 自动化工作流

## 项目概述

### 项目目标
创建一个自动化工作流系统，实现：
- 复杂任务的自动拆解
- 子任务的并行/串行执行
- 执行结果汇总和报告

### 技术架构

```
用户输入 → Mission分析 → 任务拆解 → 并行执行 → 结果汇总 → 报告
```

### 技术栈

| 组件 | 用途 | 涉及章节 |
|------|------|---------|
| Mission 模式 | 自主迭代式执行 | 2.10, 29 |
| MultiAgent | 多智能体协作 | 2.10 |
| CronManager | 定时触发 | 2.12 |
| Memory | 中间结果存储 | 2.3 |

#### 2.2 任务定义

**文件**: `workflow/task_registry.py`

```python
"""任务注册表"""

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any, Awaitable

@dataclass
class TaskResult:
    """任务结果"""
    task_id: str
    status: str  # success, failed, skipped
    output: Any
    error: str = None

class BaseTask(ABC):
    """任务基类"""

    task_id: str = None

    @abstractmethod
    async def execute(self, context: dict) -> TaskResult:
        """执行任务"""
        pass

    @abstractmethod
    def get_dependencies(self) -> list[str]:
        """获取依赖的任务"""
        pass

# 任务实现示例
class FetchSalesData(BaseTask):
    task_id = "fetch_sales_data"

    def get_dependencies(self) -> list[str]:
        return []

    async def execute(self, context: dict) -> TaskResult:
        # 模拟获取销售数据
        import random
        await asyncio.sleep(1)

        data = {
            "date": context.get("date", "2026-05-08"),
            "sales": random.randint(10000, 50000),
            "orders": random.randint(100, 500),
            "avg_order_value": random.randint(80, 200)
        }

        return TaskResult(
            task_id=self.task_id,
            status="success",
            output=data
        )

class AnalyzeTrends(BaseTask):
    task_id = "analyze_trends"

    def get_dependencies(self) -> list[str]:
        return ["fetch_sales_data", "fetch_user_data"]

    async def execute(self, context: dict) -> TaskResult:
        # 使用 AI 分析趋势
        sales_data = context.get("fetch_sales_data", {}).get("output", {})
        user_data = context.get("fetch_user_data", {}).get("output", {})

        prompt = f"""分析以下数据趋势：

销售数据：{sales_data}
用户数据：{user_data}

请给出趋势分析和关键洞察。"""

        analysis = await self.agent.run(prompt)

        return TaskResult(
            task_id=self.task_id,
            status="success",
            output={"analysis": analysis}
        )
```

### 步骤 3：工作流引擎

**文件**: `workflow/engine.py`

```python
"""工作流引擎"""

import asyncio
from typing import list
from dataclasses import dataclass
from datetime import datetime
from task_registry import BaseTask, TaskResult

@dataclass
class WorkflowState:
    """工作流状态"""
    workflow_id: str
    status: str  # pending, running, completed, failed
    tasks: dict[str, TaskResult]
    started_at: datetime
    completed_at: datetime = None

class WorkflowEngine:
    """工作流引擎"""

    def __init__(self, config: dict):
        self.config = config
        self.tasks: dict[str, BaseTask] = {}
        self.task_registry = {}

    def register_task(self, task: BaseTask):
        """注册任务"""
        self.tasks[task.task_id] = task

    def build_dag(self) -> dict[str, list[str]]:
        """构建任务依赖图"""
        dag = {}
        for task_id, task in self.tasks.items():
            dag[task_id] = task.get_dependencies()
        return dag

    def get_execution_order(self, dag: dict) -> list[list[str]]:
        """获取执行顺序（层级）"""
        in_degree = {k: len(v) for k, v in dag.items()}
        levels = []

        while in_degree:
            # 找出入度为0的任务
            current = [k for k, v in in_degree.items() if v == 0]

            if not current:
                raise ValueError("Circular dependency detected")

            levels.append(current)

            # 更新入度
            for task_id in current:
                for dependent in dag:
                    if task_id in dag[dependent]:
                        in_degree[dependent] -= 1
                del in_degree[task_id]

        return levels

    async def execute_task(self, task: BaseTask, context: dict) -> TaskResult:
        """执行单个任务"""
        try:
            result = await task.execute(context)
            context[task.task_id] = result
            return result
        except Exception as e:
            return TaskResult(
                task_id=task.task_id,
                status="failed",
                output=None,
                error=str(e)
            )

    async def run(self, initial_context: dict = None) -> WorkflowState:
        """运行工作流"""
        context = initial_context or {}
        dag = self.build_dag()
        levels = self.get_execution_order(dag)

        state = WorkflowState(
            workflow_id=generate_id(),
            status="running",
            tasks={},
            started_at=datetime.now()
        )

        for level in levels:
            # 并行执行同一层级的任务
            tasks = [self.tasks[task_id] for task_id in level]

            results = await asyncio.gather(
                *[self.execute_task(task, context) for task in tasks],
                return_exceptions=True
            )

            # 处理结果
            for task_id, result in zip(level, results):
                if isinstance(result, Exception):
                    state.tasks[task_id] = TaskResult(
                        task_id=task_id,
                        status="failed",
                        output=None,
                        error=str(result)
                    )
                else:
                    state.tasks[task_id] = result

                # 检查是否需要停止
                if self.config.get("fail_fast") and result.status == "failed":
                    state.status = "failed"
                    state.completed_at = datetime.now()
                    return state

        state.status = "completed"
        state.completed_at = datetime.now()
        return state

    async def run_mission(self, mission: str) -> str:
        """使用 Mission 模式运行"""
        # Mission 模式：自主迭代式执行
        current_plan = mission

        for iteration in range(self.config["mission"]["max_iterations"]):
            # 让 AI 分析任务
            response = await self.agent.run(
                f"请分析以下任务并决定下一步行动：\n{current_plan}\n"
                f"当前上下文：{self.get_context_summary()}"
            )

            # 检查是否完成
            if self.is_complete(response):
                return self.generate_report()

            # 执行下一步
            actions = self.parse_actions(response)
            await self.execute_actions(actions)

            # 更新上下文
            current_plan = response

        return "达到最大迭代次数"
```

### 步骤 4：报告生成

**文件**: `workflow/reporter.py`

```python
"""报告生成器"""

from typing import dict
from datetime import datetime

class ReportGenerator:
    """报告生成器"""

    def __init__(self, workflow_state: WorkflowState):
        self.state = workflow_state

    def generate(self) -> str:
        """生成报告"""
        report = f"""# 工作流执行报告

## 基本信息

- 工作流ID：{self.state.workflow_id}
- 开始时间：{self.state.started_at.strftime('%Y-%m-%d %H:%M:%S')}
- 结束时间：{self.state.completed_at.strftime('%Y-%m-%d %H:%M:%S') if self.state.completed_at else 'N/A'}
- 状态：{self.state.status}

## 任务执行结果

"""

        for task_id, result in self.state.tasks.items():
            status_icon = "✅" if result.status == "success" else "❌" if result.status == "failed" else "⏭️"
            report += f"""### {status_icon} {task_id}

- 状态：{result.status}
"""
            if result.output:
                report += f"- 输出：{self.format_output(result.output)}\n"
            if result.error:
                report += f"- 错误：{result.error}\n"
            report += "\n"

        return report

    def format_output(self, output: any) -> str:
        """格式化输出"""
        if isinstance(output, dict):
            return ", ".join([f"{k}: {v}" for k, v in list(output.items())[:3]])
        return str(output)[:100]

    def export_pdf(self, output_path: str):
        """导出 PDF"""
        # 使用 markdown2pdf 或其他工具
        import subprocess
        subprocess.run([
            "pandoc", "-o", output_path,
            "-f", "markdown",
            self.generate()
        ])
```

### 步骤 5：主程序

**文件**: `main.py`

```python
"""自动化工作流入口"""

import asyncio
from workflow.engine import WorkflowEngine
from workflow.task_registry import FetchSalesData, FetchUserData, AnalyzeTrends
from workflow.reporter import ReportGenerator

async def main():
    # 创建引擎
    engine = WorkflowEngine({
        "name": "daily_report",
        "mission": {
            "max_iterations": 5
        },
        "fail_fast": False
    })

    # 注册任务
    engine.register_task(FetchSalesData())
    engine.register_task(FetchUserData())
    engine.register_task(AnalyzeTrends())

    # 运行工作流
    state = await engine.run({
        "date": "2026-05-08"
    })

    # 生成报告
    reporter = ReportGenerator(state)
    report = reporter.generate()

    print(report)

    # 导出
    reporter.export_pdf("daily_report.pdf")

    return state

if __name__ == "__main__":
    asyncio.run(main())
```

```python
class WorkflowEngine:
    async def execute_task_with_retry(self, task, context, max_retries=2):
        for attempt in range(max_retries + 1):
            result = await self.execute_task(task, context)
            if result.status == "success":
                return result
            if attempt < max_retries:
                await asyncio.sleep(2 ** attempt)  # 指数退避
        return result
```
